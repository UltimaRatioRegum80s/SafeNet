const log = (...a)=>{ document.getElementById('log').textContent += a.map(x=>typeof x==='string'?x:JSON.stringify(x)).join(' ')+'\n'; }
const url = "https://<your-socket-origin>";
const socket = io(url, { transports:["websocket"], path:"/socket.io" }); // adjust path if custom

socket.on("connect", ()=>log("connect", socket.id));
socket.on("disconnect", (r)=>log("disconnect", r));
socket.on("connect_error", (e)=>log("connect_error", e.message));

/* replace 'incident:created' with your event */
socket.on("incident:created", (payload)=>log("incident:created", payload));

log("initialized, connecting to", url);
</script>

4) one defensive guardrail (no UX change)

add a 1-line fallback refresh on the map page only if no socket event has been seen in N seconds (won’t interfere with your stable sockets; it’s a safety net):

// pseudo-code inside map page
const lastSocketAt = useRef(Date.now());
useEffect(() => {
  socket.on("incident:created", (i) => {
    lastSocketAt.current = Date.now();
    upsertMarker(i);
  });
  const interval = setInterval(() => {
    if (Date.now() - lastSocketAt.current > 15000) {
      // 15s with no sockets? do a one-off fetch for the current bbox
      fetchIncidentsForBbox(currentBbox()).then(syncMarkers);
      lastSocketAt.current = Date.now(); // prevent loops
    }
  }, 5000);
  return () => clearInterval(interval);
}, []);


this does not introduce polling UI-wise; it only self-heals if a proxy drops the WS connection silently.
