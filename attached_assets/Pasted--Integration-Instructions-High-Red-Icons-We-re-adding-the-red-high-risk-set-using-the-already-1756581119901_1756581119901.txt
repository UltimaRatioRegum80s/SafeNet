We should merge/override by id so the new icons replace the old ones without duplicating types. Add this helper once:

function mergeTypes(base: IncidentType[], updates: IncidentType[]): IncidentType[] {
  const byId = new Map(base.map(t => [t.id, t]));
  for (const u of updates) byId.set(u.id, { ...byId.get(u.id), ...u });
  return Array.from(byId.values());
}


Then build the export:
