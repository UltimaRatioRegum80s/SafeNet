  cluster.on("layeradd", updateCount);
  cluster.on("layerremove", updateCount);

    cluster.off("layeradd", updateCount);

import { describe, it, expect } from "vitest";
import { toLatLng } from "@/components/MapView"; // or wherever it lives

describe("toLatLng", () => {
  it("accepts lat/lng numbers", () => {
    const ll = toLatLng({ lat: -33.9, lng: 18.4 })!;
    expect(ll.lat).toBeCloseTo(-33.9); expect(ll.lng).toBeCloseTo(18.4);
  });
  it("accepts latitude/longitude strings", () => {
    const ll = toLatLng({ latitude: "-33.9", longitude: "18.4" })!;
    expect(ll.lat).toBeCloseTo(-33.9); expect(ll.lng).toBeCloseTo(18.4);
  });
  it("swaps when lat is out of range", () => {
    const ll = toLatLng({ lat: 120, lng: -22 }); // swapped in
    expect(ll.lat).toBeCloseTo(-22);
    expect(ll.lng).toBeCloseTo(120);
  });
});

client/src/lib/dedupe.test.ts
import { describe, it, expect } from "vitest";

it("dedups by id", () => {
  const seen = new Set<string>();
  const add = (id: string) => { if (seen.has(id)) return false; seen.add(id); return true; };
  expect(add("a")).toBe(true);
  expect(add("a")).toBe(false);
  expect(add("b")).toBe(true);
});
