client/src/features/report/CategorySheets.tsx

import { useMemo, useState } from "react";
import { INCIDENT_TYPES } from "@/shared/incidentTypes";
import BottomSheet from "@/components/ui/BottomSheet";
import { useReportPanel } from "@/features/report/useReportPanel";
import clsx from "clsx";
import { AlertTriangle, Circle, Flame } from "lucide-react"; // icon hints on buttons

type Severity = "low" | "medium" | "high";
const COLORS = {
  low:    "from-yellow-400/90 to-yellow-500/90 text-slate-900",
  medium: "from-amber-500/90 to-amber-600/90 text-slate-900",
  high:   "from-red-600/90 to-red-700/90 text-white",
} as const;

function SeverityFab({ severity, label, onClick }:{
  severity: Severity; label: string; onClick: () => void;
}) {
  const icon =
    severity === "low" ? <Circle className="h-5 w-5" /> :
    severity === "medium" ? <AlertTriangle className="h-5 w-5" /> :
    <Flame className="h-5 w-5" />;
  return (
    <button
      onClick={onClick}
      className={clsx(
        "nn-glass pointer-events-auto",
        "fixed z-[55] rounded-full shadow-lg",
        "px-4 py-3 flex items-center gap-2",
        "backdrop-blur-md bg-gradient-to-br",
        COLORS[severity],
        "ring-1 ring-white/20",
        "active:scale-[0.98] transition"
      )}
      aria-label={`${label} incidents`}
    >
      {icon}<span className="font-semibold">{label}</span>
    </button>
  );
}

function SeverityGrid({ severity, onPick }:{
  severity: Severity; onPick: (id: string) => void;
}) {
  const items = useMemo(
    () => INCIDENT_TYPES.filter(t => t.severity === severity),
    [severity]
  );
  return (
    <div className="grid grid-cols-3 gap-3 pb-2">
      {items.map(t => (
        <button
          key={t.id}
          onClick={() => onPick(t.id)}
          className="glass-card rounded-xl p-3 flex flex-col items-center gap-2 hover:bg-white/5 active:scale-[0.98] transition"
        >
          <img
            src={t.buttonIcon ?? "/attached_assets/default.png"}
            alt={t.label}
            className="h-10 w-10"
            loading="lazy"
            decoding="async"
          />
          <span className="text-xs text-center">{t.label}</span>
        </button>
      ))}
    </div>
  );
}

export default function CategorySheets() {
  const [open, setOpen] = useState<Severity | null>(null);
  const rp = useReportPanel();

  const onPick = (id: string) => {
    setOpen(null);
    rp.openWith(id); // opens ReportPanel with this type
  };

  // FAB placement (bottom-right cluster, safe area aware)
  const basePos = "bottom-[calc(env(safe-area-inset-bottom)+88px)] right-4 sm:right-6";

  return (
    <>
      {/* FABs: mobile/tablet only */}
      <div className="fixed inset-0 pointer-events-none sm:block lg:hidden">
        <div className="pointer-events-none">
          <SeverityFab
            severity="low"
            label="Low"
            onClick={() => setOpen("low")}
          />
          <style jsx>{`
            button[aria-label="Low incidents"] { right: 1rem; bottom: calc(env(safe-area-inset-bottom) + 168px); }
            button[aria-label="Medium incidents"] { right: 1rem; bottom: calc(env(safe-area-inset-bottom) + 120px); }
            button[aria-label="High incidents"] { right: 1rem; bottom: calc(env(safe-area-inset-bottom) + 72px); }
            @media (min-width: 640px) {
              button[aria-label$="incidents"] { right: 1.5rem; }
            }
          `}</style>
        </div>
        <SeverityFab severity="medium" label="Medium" onClick={() => setOpen("medium")} />
        <SeverityFab severity="high" label="High" onClick={() => setOpen("high")} />
      </div>

      {/* Sheets */}
      <BottomSheet
        open={open === "low"}
        onOpenChange={(v) => setOpen(v ? "low" : null)}
        title="Low-risk incidents"
        snapPoints={[0.45, 0.8, 1]}
        initialSnap={1}
      >
        <SeverityGrid severity="low" onPick={onPick} />
      </BottomSheet>

      <BottomSheet
        open={open === "medium"}
        onOpenChange={(v) => setOpen(v ? "medium" : null)}
        title="Medium-risk incidents"
        snapPoints={[0.45, 0.8, 1]}
        initialSnap={1}
      >
        <SeverityGrid severity="medium" onPick={onPick} />
      </BottomSheet>

      <BottomSheet
        open={open === "high"}
        onOpenChange={(v) => setOpen(v ? "high" : null)}
        title="High-risk incidents"
        snapPoints={[0.45, 0.8, 1]}
        initialSnap={1}
      >
        <SeverityGrid severity="high" onPick={onPick} />
      </BottomSheet>
    </>
  );
}
