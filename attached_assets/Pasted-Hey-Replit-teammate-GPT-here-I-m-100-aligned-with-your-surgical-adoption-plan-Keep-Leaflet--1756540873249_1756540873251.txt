// client/src/lib/timeAgo.ts
export function timeAgo(iso?: string) {
  if (!iso) return "Just now";
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return "Unknown time";
  const s = Math.floor((Date.now() - d.getTime()) / 1000);
  if (s < 60) return `${s}s ago`;
  const m = Math.floor(s / 60);
  if (m < 60) return `${m}m ago`;
  const h = Math.floor(m / 60);
  if (h < 24) return `${h}h ago`;
  const days = Math.floor(h / 24);
  return `${days}d ago`;
}

export function SeverityChip({ sev }: { sev: Severity }) {
  return (
    <span className="px-2 py-0.5 rounded-full bg-white/10 border border-white/10 text-[10px] font-semibold">
      {sev.toUpperCase()}
    </span>
  );
}

export function SeverityIcon({ sev, className="w-4 h-4 text-white/80" }:{ sev: Severity; className?: string }) {
  switch (sev) {
    case "emergency": return <Siren className={className} />;
    case "high": return <AlertTriangle className={className} />;
    case "medium": return <MapPin className={className} />;
    default: return <HandHelping className={className} />;
  }
}

D) Vaul sheet “peek/half/full” behavior (without new deps)

In your existing IncidentSheet (vaul Drawer):

// top-level in component
const [height, setHeight] = React.useState<"peek"|"half"|"full">("peek");

// in Content wrapper
<Drawer.Content className="fixed inset-x-0 bottom-0 z-30">
  <div className={`mx-auto max-w-md transition-[height] duration-200
    ${height === "full" ? "h-[80vh]" : height === "half" ? "h-[52vh]" : "h-[96px]"}
  `}>
    {/* rest of the sheet */}
  </div>
</Drawer.Content>

// handle element (tap to cycle)
<button onClick={() => setHeight(h => (h === "peek" ? "half" : h === "half" ? "full" : "peek"))}>
  <div className="h-1.5 w-12 rounded-full bg-white/30 mx-auto mt-2" />
</button>


This gives you a simple cycle between peek → half → full. If you want true drag physics later, we can add it, but this is minimal and robust on mobile.
