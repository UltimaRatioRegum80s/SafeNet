We don’t need it. Here’s a drop-in version of MobileMapChrome.tsx without clsx, using plain class strings and improved touch behavior:

// client/src/components/map/MobileMapChrome.tsx
import React from "react";
import { Home, FilePlus2, Newspaper, Minus, Plus, Sun, Moon } from "lucide-react";

type Props = {
  onZoomIn: () => void;
  onZoomOut: () => void;
  mode: "light" | "dark";
  onToggleMode: () => void;
  onNav?: (dest: "home" | "report" | "feed") => void;
};

export default function MobileMapChrome({
  onZoomIn,
  onZoomOut,
  mode,
  onToggleMode,
  onNav,
}: Props) {
  // wrapper uses pointer-events-none so the map remains draggable except on buttons
  return (
    <div className="md:hidden pointer-events-none">
      {/* Top Nav */}
      <div
        className={
          [
            "fixed left-3 right-3 z-[60] pointer-events-auto",
            "rounded-2xl shadow-md border",
            "backdrop-blur bg-background/70",
            "flex items-center justify-between px-2 py-1",
            "mt-[max(env(safe-area-inset-top),0.5rem)]",
          ].join(" ")
        }
      >
        <button
          className="flex items-center gap-2 px-3 py-2 rounded-xl text-sm hover:bg-muted/60 active:scale-95 transition"
          onClick={() => onNav?.("home")}
        >
          <Home size={18} /> Home
        </button>
        <button
          className="flex items-center gap-2 px-3 py-2 rounded-xl text-sm hover:bg-muted/60 active:scale-95 transition"
          onClick={() => onNav?.("report")}
        >
          <FilePlus2 size={18} /> Report
        </button>
        <button
          className="flex items-center gap-2 px-3 py-2 rounded-xl text-sm hover:bg-muted/60 active:scale-95 transition"
          onClick={() => onNav?.("feed")}
        >
          <Newspaper size={18} /> Feed
        </button>
      </div>

      {/* Right stack: zoom + theme */}
      <div className="fixed right-3 bottom-28 z-[60] flex flex-col gap-2 pointer-events-auto">
        <button
          aria-label="Zoom in"
          className="h-12 w-12 grid place-items-center rounded-2xl bg-background/90 border shadow-md active:scale-95 transition"
          onClick={onZoomIn}
        >
          <Plus size={20} />
        </button>
        <button
          aria-label="Zoom out"
          className="h-12 w-12 grid place-items-center rounded-2xl bg-background/90 border shadow-md active:scale-95 transition"
          onClick={onZoomOut}
        >
          <Minus size={20} />
        </button>
        <button
          aria-label="Toggle theme"
          className="h-12 w-12 grid place-items-center rounded-2xl bg-background/90 border shadow-md active:scale-95 transition"
          onClick={onToggleMode}
        >
          {mode === "light" ? <Moon size={20} /> : <Sun size={20} />}
        </button>
      </div>
    </div>
  );
}
