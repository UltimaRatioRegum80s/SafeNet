Step 2: Add the Shared Component

Use a single shared navigation component across all pages.

📄 src/components/FloatingNav.tsx
import { Home, Map, FileText, Rss } from "lucide-react";
import { useLocation, useNavigate } from "react-router-dom";
import { motion } from "framer-motion";
import { cn } from "@/lib/utils";

export default function FloatingNav() {
  const navigate = useNavigate();
  const location = useLocation();

  const navItems = [
    { name: "Home", icon: Home, path: "/" },
    { name: "Map", icon: Map, path: "/map" },
    { name: "Report", icon: FileText, path: "/report" },
    { name: "Feed", icon: Rss, path: "/feed" },
  ];

  return (
    <motion.nav
      initial={{ y: -30, opacity: 0 }}
      animate={{ y: 0, opacity: 1 }}
      transition={{ duration: 0.3, ease: "easeOut" }}
      className={cn(
        "fixed top-3 left-1/2 z-50 -translate-x-1/2 flex items-center justify-center gap-2",
        "rounded-2xl bg-neutral-900/80 backdrop-blur-md shadow-lg border border-white/10 px-3 py-2",
        "md:hidden" // mobile only
      )}
    >
      {navItems.map(({ name, icon: Icon, path }) => {
        const isActive = location.pathname === path;
        return (
          <button
            key={name}
            onClick={() => navigate(path)}
            className={cn(
              "flex flex-col items-center justify-center px-4 py-2 rounded-xl text-xs font-medium transition-all duration-200",
              isActive
                ? "bg-blue-500/20 text-blue-400"
                : "text-gray-300 hover:bg-white/10 hover:text-white"
            )}
          >
            <Icon size={18} strokeWidth={2} />
          </button>
        );
      })}
    </motion.nav>
  );
}

✅ Key Features

Works identically on all pages (Home, Map, Report, Feed).

Uses react-router’s useNavigate for seamless routing.

Active route is highlighted in blue.

Has Framer Motion animation for smooth appearance.

Limited to mobile view only (md:hidden).

Safe integration — no interference with map or backend logic.
