const COOLDOWN_MS = 30_000;
const COOLDOWN_KEY = "nn:lastIncidentSubmitAt";

export default function ReportPanel() {
  // replace lastSubmitAt init:
  const [lastSubmitAt, setLastSubmitAt] = useState<number | null>(() => {
    const v = localStorage.getItem(COOLDOWN_KEY);
    return v ? Number(v) : null;
  });
  const [now, setNow] = useState(() => Date.now());

  useEffect(() => {
    const tick = () => setNow(Date.now());
    const i = setInterval(tick, 500);
    window.addEventListener("focus", tick);
    return () => { clearInterval(i); window.removeEventListener("focus", tick); };
  }, []);

  const msLeft = Math.max(0, (lastSubmitAt ?? 0) + COOLDOWN_MS - now);
  const cooldownActive = msLeft > 0;

      const ts = Date.now();
      setLastSubmitAt(ts);

// client/src/store/incidentStore.ts
type Incident = {
  id: string;
  type: string;
  status: "pending" | "approved" | "rejected";
  lat?: number; lng?: number; createdAt: string;
  // ...whatever else you expose to map/feed
};

type State = {
  incidents: Incident[];
  addIncident: (i: Incident) => void;
  reconcileIncident: (tempId: string, real: Incident) => void;
};

export const useIncidentStore = create<State>((set) => ({
  incidents: [],
  addIncident: (i) => set((s) => ({ incidents: [i, ...s.incidents] })),
  reconcileIncident: (tempId, real) =>
    set((s) => ({ incidents: s.incidents.map(x => x.id === tempId ? real : x) })),
}));

socket.on("incident:created", (incident) => {
  // if the optimistic entry exists, reconcile; else add fresh
  const state = useIncidentStore.getState();
  const temp = state.incidents.find(i =>
    i.id.startsWith("temp-") && i.type === incident.type &&
    Math.hypot((i.lat ?? 0) - (incident.lat ?? 0), (i.lng ?? 0) - (incident.lng ?? 0)) < 0.0005
  );
  temp ? state.reconcileIncident(temp.id, incident) : state.addIncident(incident);
});
