2) Disable submit while locating/posting (race-proof)
const [isLocating, setIsLocating] = React.useState(false);
const isBusy = isLocating || createIncident.isPending;

const onSubmit = async () => {
  if (!type || !severity || isBusy) return;
  try {
    setIsLocating(true);
    const { lat, lng } = await getCurrentPosition();
    await createIncident.mutateAsync({
      type, severity, latitude: lat, longitude: lng,
      description: note?.trim() || undefined,
    });
    reset();
    toast({ title: "Incident reported", description: "Redirecting to map…" });
    setLocation("/map");
  } catch (err: any) {
    toast({
      title: "Couldn’t get your location",
      description: err?.message ?? "Please enable location and try again.",
      variant: "destructive",
    });
  } finally {
    setIsLocating(false);
  }
};

// ...
<Button onClick={onSubmit} disabled={isBusy || !type || !severity}>
  {isBusy ? "Submitting…" : "Submit"}
</Button>
