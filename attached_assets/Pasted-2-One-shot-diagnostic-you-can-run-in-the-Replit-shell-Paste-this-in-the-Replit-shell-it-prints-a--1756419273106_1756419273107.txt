4) Drop-in “old look, new alignment” bubble component (ready to paste)

Use this as src/components/Chat/ChatBubble.tsx so we can hook it up immediately:

import React from "react";
import clsx from "clsx";

type MessageProps = {
  text: string;
  isUser?: boolean;
  timestamp?: string;
};

export default function ChatBubble({ text, isUser, timestamp }: MessageProps) {
  return (
    <div className={clsx("flex w-full mb-2", isUser ? "justify-end" : "justify-start")}>
      <div
        className={clsx(
          "max-w-[75%] rounded-2xl px-4 py-2 text-sm shadow-md",
          "transition-colors",
          isUser
            ? "bg-teal-600 text-white rounded-br-none"
            : "bg-zinc-800 text-zinc-100 rounded-bl-none"
        )}
      >
        <p className="whitespace-pre-wrap break-words">{text}</p>
        {timestamp && (
          <p className="text-[10px] opacity-70 mt-1 text-right">{timestamp}</p>
        )}
      </div>
    </div>
  );
}


And a quick list item that uses it:

// src/components/Chat/MessageItem.tsx
import ChatBubble from "./ChatBubble";

export function MessageItem({ m }: { m: { id: string; text: string; userId: string; createdAt?: string } }) {
  const me = typeof window !== "undefined" ? localStorage.getItem("uid") : null;
  const isUser = m.userId === me;
  return (
    <ChatBubble
      text={m.text}
      isUser={isUser}
      timestamp={m.createdAt ? new Date(m.createdAt).toLocaleTimeString() : undefined}
    />
  );
}
