import React, { useMemo, useRef, useState } from "react";
import { Camera, Paperclip, Send, Plus, Smile, AlertTriangle } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Tabs, TabsList, TabsTrigger, TabsContent } from "@/components/ui/tabs";
import { Drawer, DrawerContent, DrawerHeader, DrawerTitle, DrawerFooter } from "@/components/ui/drawer";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { Textarea } from "@/components/ui/textarea";
import { Input } from "@/components/ui/input";
import clsx from "clsx";

/** --- TYPES --- */
type FeedFilter = "all" | "incidents" | "chat";
type Severity = "low" | "medium" | "high";

type BaseItem = {
  id: string;
  createdAt: string; // ISO
};
type ChatItem = BaseItem & {
  kind: "chat";
  text: string;
  mine?: boolean;
  anonymous?: boolean;
};
type IncidentItem = BaseItem & {
  kind: "incident";
  title: string;
  description?: string;
  severity: Severity;
  attachments?: string[];
};

type FeedItem = ChatItem | IncidentItem;

/** --- MOCK DATA (replace with API) --- */
const demoItems: FeedItem[] = [
  { id: "1", kind: "chat", text: "Hello neighbors 👋", mine: false, createdAt: new Date().toISOString() },
  { id: "2", kind: "incident", title: "Dog found near the beach", description: "Brown lab, no collar.", severity: "low", createdAt: new Date().toISOString() },
  { id: "3", kind: "chat", text: "Testing real-time chat", mine: true, createdAt: new Date().toISOString() },
  { id: "4", kind: "incident", title: "Suspicious noise last night", description: "Around 2am near Vine St.", severity: "medium", createdAt: new Date().toISOString() },
];

/** --- UTIL --- */
const timeShort = (iso: string) =>
  new Intl.DateTimeFormat(undefined, { hour: "2-digit", minute: "2-digit" }).format(new Date(iso));

/** --- MESSAGE BUBBLE (WhatsApp-style) --- */
function ChatBubble({ item }: { item: ChatItem }) {
  const mine = !!item.mine;
  return (
    <div className={clsx("mb-1 px-2 w-full flex", mine ? "justify-end" : "justify-start")}>
      <div
        className={clsx(
          "max-w-[82%] rounded-2xl px-3 py-2 text-sm leading-snug",
          mine ? "bg-[#4cc0ff]/90 text-white rounded-br-sm" : "bg-zinc-800 text-zinc-100 rounded-bl-sm"
        )}
      >
        <div className="whitespace-pre-wrap break-words">{item.text}</div>
        <div className={clsx("mt-0.5 text-[10px]", mine ? "text-white/80" : "text-zinc-400")}>{timeShort(item.createdAt)}</div>
      </div>
    </div>
  );
}

/** --- INCIDENT CARD (compact, color by severity) --- */
const severityStyles: Record<Severity, string> = {
  low: "border-emerald-500/40 bg-emerald-500/10",
  medium: "border-amber-500/40 bg-amber-500/10",
  high: "border-rose-500/40 bg-rose-500/10",
};

function IncidentCard({ item }: { item: IncidentItem }) {
  return (
    <div className="px-2 mb-2">
      <div className={clsx("rounded-2xl border p-3 text-sm", severityStyles[item.severity])}>
        <div className="flex items-center gap-2 font-medium">
          <AlertTriangle className="h-4 w-4 opacity-80" />
          <span>{item.title}</span>
          <span
            className={clsx(
              "ml-auto text-[10px] px-2 py-0.5 rounded-full",
              item.severity === "high" ? "bg-rose-500 text-white" :
              item.severity === "medium" ? "bg-amber-500 text-black" : "bg-emerald-500 text-black"
            )}
          >
            {item.severity.toUpperCase()}
          </span>
        </div>
        {item.description && <div className="mt-1 text-zinc-300">{item.description}</div>}
        <div className="mt-1 text-[10px] text-zinc-400">{timeShort(item.createdAt)}</div>
      </div>
    </div>
  );
}

/** --- MAIN --- */
export default function CommunityFeedMobile() {
  const [filter, setFilter] = useState<FeedFilter>("all");
  const [input, setInput] = useState("");
  const [drawerOpen, setDrawerOpen] = useState(false);

  const fileCameraRef = useRef<HTMLInputElement>(null);
  const fileAttachRef = useRef<HTMLInputElement>(null);

  const feed = useMemo(() => {
    switch (filter) {
      case "incidents": return demoItems.filter(i => i.kind === "incident");
      case "chat": return demoItems.filter(i => i.kind === "chat");
      default: return demoItems;
    }
  }, [filter]);

  const handleSend = () => {
    if (!input.trim()) return;
    // TODO: socket emit → add chat message
    setInput("");
  };

  const handleOpenCamera = () => fileCameraRef.current?.click();
  const handleOpenAttach = () => fileAttachRef.current?.click();

  return (
    <div className="flex flex-col h-[100dvh] bg-[#0b1220] text-white">
      {/* Top bar */}
      <div className="sticky top-0 z-20 border-b border-white/5 bg-[#0b1220]/80 backdrop-blur">
        <div className="px-3 py-2 flex items-center gap-2">
          <div className="font-semibold text-base">Community Feed</div>
        </div>

        {/* Segmented tabs */}
        <Tabs value={filter} onValueChange={(v) => setFilter(v as FeedFilter)} className="px-2 pb-2">
          <TabsList className="grid grid-cols-3 w-full bg-zinc-900">
            <TabsTrigger value="all">All</TabsTrigger>
            <TabsTrigger value="incidents">Incidents</TabsTrigger>
            <TabsTrigger value="chat">Chat</TabsTrigger>
          </TabsList>
          {/* We don’t render per-tab content here; list is filtered below */}
          <TabsContent value="all" />
          <TabsContent value="incidents" />
          <TabsContent value="chat" />
        </Tabs>
      </div>

      {/* Messages list */}
      <div className="flex-1 overflow-y-auto pb-24 pt-2">
        {feed.map((item) =>
          item.kind === "chat" ? (
            <ChatBubble key={item.id} item={item} />
          ) : (
            <IncidentCard key={item.id} item={item} />
          )
        )}
      </div>

      {/* Bottom input bar (WhatsApp-style) */}
      <div className="sticky bottom-0 z-30 bg-[#0b1220]/95 backdrop-blur border-t border-white/5">
        <div className="px-2 py-2 flex items-end gap-2">
          <Button variant="ghost" size="icon" className="h-10 w-10 rounded-full" onClick={() => {/* TODO emoji picker */}}>
            <Smile className="h-5 w-5" />
          </Button>

          {/* Hidden camera input triggers phone camera */}
          <input
            ref={fileCameraRef}
            type="file"
            accept="image/*"
            capture="environment"
            className="hidden"
            onChange={(e) => {
              const file = e.target.files?.[0];
              if (!file) return;
              // TODO: create incident draft w/ image or attach to message
              setDrawerOpen(true);
            }}
          />
          <Button variant="ghost" size="icon" className="h-10 w-10 rounded-full" onClick={handleOpenCamera}>
            <Camera className="h-5 w-5" />
          </Button>

          {/* Hidden attachment picker */}
          <input
            ref={fileAttachRef}
            type="file"
            className="hidden"
            multiple
            onChange={(e) => {
              const files = e.target.files;
              if (!files?.length) return;
              // TODO: upload & attach
            }}
          />
          <Button variant="ghost" size="icon" className="h-10 w-10 rounded-full" onClick={handleOpenAttach}>
            <Paperclip className="h-5 w-5" />
          </Button>

          <div className="flex-1">
            <Input
              value={input}
              onChange={(e) => setInput(e.target.value)}
              placeholder="Type a message…"
              className="bg-zinc-900 border-zinc-800 text-sm rounded-full px-4 h-10"
            />
          </div>

          <Button onClick={handleSend} className="h-10 rounded-full bg-cyan-600 hover:bg-cyan-500">
            <Send className="h-4 w-4" />
          </Button>
        </div>

        {/* Safe-area spacer */}
        <div style={{ height: "calc(env(safe-area-inset-bottom))" }} />
      </div>

      {/* FAB: Report Incident */}
      <Button
        onClick={() => setDrawerOpen(true)}
        className="fixed bottom-20 right-4 h-14 w-14 rounded-full bg-cyan-600 shadow-lg shadow-cyan-600/30 hover:bg-cyan-500"
      >
        <Plus className="h-6 w-6" />
      </Button>

      {/* Mobile Drawer for Incident Report */}
      <Drawer open={drawerOpen} onOpenChange={setDrawerOpen}>
        <DrawerContent className="bg-[#0b1220] text-white border-t border-white/10">
          <DrawerHeader>
            <DrawerTitle>Report Incident</DrawerTitle>
          </DrawerHeader>
          <div className="px-4 space-y-3">
            <Select defaultValue="medium">
              <SelectTrigger className="bg-zinc-900 border-zinc-800">
                <SelectValue placeholder="Severity" />
              </SelectTrigger>
              <SelectContent>
                <SelectItem value="low">Low</SelectItem>
                <SelectItem value="medium">Medium</SelectItem>
                <SelectItem value="high">High</SelectItem>
              </SelectContent>
            </Select>

            <Input placeholder="Title" className="bg-zinc-900 border-zinc-800" />
            <Textarea placeholder="Describe what happened…" className="bg-zinc-900 border-zinc-800 min-h-[120px]" />
            {/* TODO: image preview(s) if captured/attached */}
          </div>
          <DrawerFooter>
            <Button className="bg-cyan-600 hover:bg-cyan-500">Submit</Button>
            <Button variant="outline" className="border-zinc-700" onClick={() => setDrawerOpen(false)}>Cancel</Button>
          </DrawerFooter>
        </DrawerContent>
      </Drawer>
    </div>
  );
}
