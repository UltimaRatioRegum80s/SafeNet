  const [ok, setOk] = useState(true);
  return ok ? (
    <img
      src={src} alt={alt} loading="lazy" decoding="async"

2) Unified image component (lazy, fade-in, fallback)

New File: client/src/components/NNImage.tsx

import { useState } from "react";
import clsx from "clsx";

export default function NNImage({
  src, alt, className, onClick,
}: { src: string; alt: string; className?: string; onClick?: () => void }) {
  const [loaded, setLoaded] = useState(false);
  const [ok, setOk] = useState(true);

  if (!ok) {
    return (
      <div
        role="img" aria-label="image unavailable"
        className={clsx("rounded-lg bg-slate-800/60 border border-slate-700/60", className)}
      />
    );
  }

  return (
    <img
      src={src}
      alt={alt}
      loading="lazy"
      decoding="async"
      onLoad={() => setLoaded(true)}
      onError={() => setOk(false)}
      onClick={onClick}
      className={clsx(
        "rounded-lg object-cover transition-opacity duration-300",
        loaded ? "opacity-100" : "opacity-0",
        className
      )}
    />
  );
}
