1) MapView.tsx — install a topmost “simple dots” layer + severity colors

Place this right after the map instance is created (and before any data render):

// === Simple severity dots (always on, sits above everything) ===
const nnPane = map.createPane("nnTop");
nnPane.style.zIndex = "650";          // above markerPane(600)
nnPane.style.pointerEvents = "none";

(function ensureSimpleDots() {
  const w = window as any;
  if (!w.__nnSimpleDots) {
    const group = L.layerGroup();
    const byId = new Map<string, L.CircleMarker>();

    const severityColor: Record<string, string> = {
      low: "#facc15",      // yellow
      medium: "#f59e0b",   // amber
      high: "#ef4444",     // red
      critical: "#b91c1c", // darker red
    };

    function add(id: string, lat: number, lng: number, severity: string = "low") {
      if (byId.has(id)) return;
      const color = severityColor[severity] || "#f59e0b";
      const m = L.circleMarker([lat, lng], {
        pane: "nnTop",
        radius: 9,
        color,
        weight: 2,
        fillColor: color,
        fillOpacity: 0.95,
      }).addTo(group);
      byId.set(id, m);
    }

    function remove(id: string) {
      const m = byId.get(id);
      if (!m) return;
      group.removeLayer(m);
      byId.delete(id);
    }

    w.__nnSimpleDots = { group, add, remove };
  }
  const dots = (window as any).__nnSimpleDots;
  if (!map.hasLayer(dots.group)) dots.group.addTo(map);
})();


This does not touch your cluster; it only ensures we always have a visible layer for severity-colored dots.
