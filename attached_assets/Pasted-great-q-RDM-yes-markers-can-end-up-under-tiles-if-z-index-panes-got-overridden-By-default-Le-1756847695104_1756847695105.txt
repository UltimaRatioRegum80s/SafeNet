great q, @RDM — yes, markers can end up “under” tiles if z-index/panes got overridden. By default Leaflet prevents this (tile-pane z=200; marker-pane z=600), so if they’re hidden it’s almost always custom CSS.

// Compare pane z-index/visibility
(() => {
  const panes = ['tile','overlay','shadow','marker','tooltip','popup'];
  const res = {};
  panes.forEach(k => {
    const el = document.querySelector(`.leaflet-${k}-pane`);
    res[k] = el ? {
      exists: true,
      z: getComputedStyle(el).zIndex,
      display: getComputedStyle(el).display,
      opacity: getComputedStyle(el).opacity
    } : { exists:false };
  });
  console.table(res); return res;
})();
