Unified Image component

export function NNImage({ src, alt, className }: {src:string; alt:string; className?:string}) {
  const [ok, setOk] = useState(true);
  return ok ? (
    <img
      src={src} alt={alt} loading="lazy" decoding="async"
      onError={() => setOk(false)}
      className={clsx("rounded-lg object-cover transition-opacity duration-300", className)}
    />
  ) : (
    <div className={clsx("rounded-lg bg-slate-800/60 border border-slate-700/60", className)} aria-label="image unavailable" />
  );
}
