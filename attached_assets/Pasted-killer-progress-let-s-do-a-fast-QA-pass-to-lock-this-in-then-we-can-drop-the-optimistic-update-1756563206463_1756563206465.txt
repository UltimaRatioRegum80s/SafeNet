1) Extend your existing useIncidentStore

Add two minimal methods; keep your structure unchanged elsewhere.

// client/src/store/incidentStore.ts
import { create } from "zustand";

type Incident = {
  id: string;
  type: string;
  status: "pending" | "approved" | "rejected";
  lat?: number; lng?: number;
  createdAt: string;
  // ...your other fields
};

type State = {
  incidents: Incident[];
  addIncident: (i: Incident) => void;
  reconcileIncident: (tempId: string, real: Incident) => void;
};

export const useIncidentStore = create<State>()((set, get) => ({
  incidents: get()?.incidents ?? [],
  addIncident: (i) => set((s) => ({ incidents: [i, ...(s.incidents ?? [])] })),
  reconcileIncident: (tempId, real) =>
    set((s) => ({
      incidents: (s.incidents ?? []).map(x => x.id === tempId ? real : x),
    })),
}));

2) Optimistic add in ReportPanel

Insert right before mutation.mutate(...) and reconcile on success.

// client/src/features/report/ReportPanel.tsx
import { useIncidentStore } from "@/store/incidentStore";

// inside component
const addIncident = useIncidentStore((s) => s.addIncident);
const reconcileIncident = useIncidentStore((s) => s.reconcileIncident);

function handleSubmit() {
  if (!selected || cooldownActive || !hasCoords) return;

  const tempId = `temp-${crypto.randomUUID()}`;
  addIncident({
    id: tempId,
    type: selected.id,
    status: "pending",
    lat: chosenCoords?.lat,
    lng: chosenCoords?.lng,
    createdAt: new Date().toISOString(),
  });

  mutation.mutate({
    type: selected.id,
    severity: selected.severity,
    description: description.trim() || undefined,
    isAnonymous,
    lat: chosenCoords?.lat,
    lng: chosenCoords?.lng,
    photos: photos.length ? photos.slice(0, 2) : undefined,
  }, {
    onSuccess: (resp) => {
      reconcileIncident(tempId, resp.incident);
    }
  });
}

3) Reconcile on socket receive (in your socket bootstrap)

If the optimistic one exists, replace; otherwise add fresh.

// client/src/lib/realtime.ts (or where you init sockets)
import { useIncidentStore } from "@/store/incidentStore";
import { socket } from "./socket"; // your existing client

socket.on("incident:created", (incident) => {
  const state = useIncidentStore.getState();
  const temp = state.incidents.find(i =>
    i.id.startsWith("temp-") &&
    i.type === incident.type &&
    Math.hypot((i.lat ?? 0) - (incident.lat ?? 0), (i.lng ?? 0) - (incident.lng ?? 0)) < 0.0007
  );
  temp ? state.reconcileIncident(temp.id, incident) : state.addIncident(incident);
});
