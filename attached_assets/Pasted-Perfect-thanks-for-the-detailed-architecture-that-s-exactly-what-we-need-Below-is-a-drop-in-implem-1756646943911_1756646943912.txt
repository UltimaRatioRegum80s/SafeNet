1) Geolocation helper

client/src/lib/geolocation.ts

// client/src/lib/geolocation.ts
export type GeoPoint = { lat: number; lng: number; accuracy?: number };

export class GeolocationError extends Error {
  code?: number;
  constructor(message: string, code?: number) {
    super(message);
    this.name = "GeolocationError";
    this.code = code;
  }
}

export function getCurrentPosition(
  opts: Partial<PositionOptions> = {}
): Promise<GeoPoint> {
  const options: PositionOptions = {
    enableHighAccuracy: true,
    timeout: 8000,
    maximumAge: 0,
    ...opts,
  };

  return new Promise((resolve, reject) => {
    if (!("geolocation" in navigator)) {
      return reject(new GeolocationError("Geolocation not supported"));
    }
    const onSuccess = (pos: GeolocationPosition) => {
      resolve({
        lat: pos.coords.latitude,
        lng: pos.coords.longitude,
        accuracy: pos.coords.accuracy,
      });
    };
    const onError = (err: GeolocationPositionError) => {
      reject(new GeolocationError(err.message, err.code));
    };
    navigator.geolocation.getCurrentPosition(onSuccess, onError, options);
  });
}

  isOpen: false,
  type: undefined,
  severity: undefined,
  note: "",
  open: ({ type, severity }) => set({ isOpen: true, type, severity, note: "" }),
  close: () => set({ isOpen: false }),
  setNote: (note) => set({ note }),
  reset: () => set({ isOpen: false, type: undefined, severity: undefined, note: "" }),
}));

3) React Query mutation for creating an incident

client/src/features/report/api/useCreateIncident.ts

// client/src/features/report/api/useCreateIncident.ts
import { useMutation, useQueryClient } from "@tanstack/react-query";
import type { IncidentType, IncidentSeverity } from "@/features/report/incidentTypes";

export type CreateIncidentDTO = {
  type: IncidentType;
  severity: IncidentSeverity;
  latitude: number;
  longitude: number;
  description?: string;
};

async function postIncident(dto: CreateIncidentDTO) {
  const res = await fetch("/api/incidents", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(dto),
  });
  if (!res.ok) {
    const text = await res.text().catch(() => "");
    throw new Error(text || `Failed to create incident (${res.status})`);
  }
  return (await res.json()) as { id: string };
}

export function useCreateIncident() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: postIncident,
    onSuccess: () => {
      // refresh any lists/feeds if present
      qc.invalidateQueries({ queryKey: ["incidents"] }).catch(() => {});
    },
  });
}

4) QuickReportCard bottom sheet

client/src/features/report/QuickReportCard.tsx

// client/src/features/report/QuickReportCard.tsx
import * as React from "react";
import { Sheet, SheetContent, SheetHeader, SheetTitle, SheetFooter } from "@/components/ui/sheet";
import { Button } from "@/components/ui/button";
import { Textarea } from "@/components/ui/textarea";
import { useQuickReportStore } from "@/state/useQuickReportStore";
import { getCurrentPosition } from "@/lib/geolocation";
import { useCreateIncident } from "./api/useCreateIncident";
import { useLocation } from "wouter";
import { useToast } from "@/components/ui/use-toast";
import { INCIDENT_TYPES } from "@/features/report/incidentTypes";

export default function QuickReportCard() {
  const { isOpen, type, severity, note, setNote, close, reset } = useQuickReportStore();
  const { toast } = useToast?.() ?? { toast: (_: any) => {} }; // safe if toast not installed
  const [, setLocation] = useLocation();
  const createIncident = useCreateIncident();

  const label =
    (type && INCIDENT_TYPES?.[type]?.label) ||
    (type ? type.replace(/_/g, " ") : "Report");

  const onSubmit = async () => {
    if (!type || !severity) return;

    try {
      const { lat, lng } = await getCurrentPosition();
      await createIncident.mutateAsync({
        type,
        severity,
        latitude: lat,
        longitude: lng,
        description: note?.trim() || undefined,
      });

      // Let the socket/server broadcast; redirect user to map to see the pin.
      reset();
      toast?.({ title: "Incident reported", description: "Redirecting to map…" });
      setLocation("/map");
    } catch (err: any) {
      toast?.({
        title: "Couldn’t get your location",
        description: err?.message ?? "Please enable location and try again.",
        variant: "destructive",
      });
    }
  };

  return (
    <Sheet open={isOpen} onOpenChange={(o) => (!o ? close() : null)}>
      <SheetContent side="bottom" className="rounded-t-2xl bg-neutral-950 text-white">
        <SheetHeader>
          <SheetTitle className="text-white">{label}</SheetTitle>
          <p className="text-xs text-white/60">
            Optional: add a short note, then submit to pin your location.
          </p>
        </SheetHeader>

        <div className="mt-4 space-y-4">
          <div className="space-y-2">
            <label className="text-sm text-white/80">Short note (optional)</label>
            <Textarea
              value={note}
              onChange={(e) => setNote(e.target.value)}
              placeholder="What happened?"
              className="min-h-[90px] resize-none bg-neutral-900 text-white"
            />
          </div>
        </div>

        <SheetFooter className="mt-6 flex items-center gap-2">
          <Button variant="ghost" className="text-white/70" onClick={close}>
            Cancel
          </Button>
          <Button
            className="ml-auto bg-teal-500 text-black hover:bg-teal-400"
            onClick={onSubmit}
            disabled={createIncident.isPending || !type || !severity}
          >
            {createIncident.isPending ? "Submitting…" : "Submit"}
          </Button>
        </SheetFooter>
      </SheetContent>
    </Sheet>
  );
}
