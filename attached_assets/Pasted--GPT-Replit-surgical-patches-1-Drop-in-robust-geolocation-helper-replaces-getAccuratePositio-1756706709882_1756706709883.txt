1) Drop-in robust geolocation helper (replaces getAccuratePosition)

Create client/src/lib/location.ts:

// client/src/lib/location.ts
export type Coords = { lat: number; lng: number; accuracy?: number; source?: string };

const LS_KEY = "nn:lastKnownCoords";

function saveLastKnown(c: Coords) {
  try { localStorage.setItem(LS_KEY, JSON.stringify({ ...c, t: Date.now() })); } catch {}
}
export function loadLastKnown(maxAgeMs = 5 * 60 * 1000): Coords | null {
  try {
    const raw = localStorage.getItem(LS_KEY);
    if (!raw) return null;
    const obj = JSON.parse(raw);
    if (Date.now() - obj.t > maxAgeMs) return null;
    return { lat: obj.lat, lng: obj.lng, accuracy: obj.accuracy, source: "lastKnown" };
  } catch { return null; }
}

/**
 * Adaptive geolocation:
 * - Races a high-accuracy watch against a quick coarse getCurrentPosition
 * - Resolves early if targetAccuracy met, or returns best-so-far at maxTime
 * - Never leaves you empty-handed: falls back to lastKnown, then center
 */
export function getAdaptivePosition(opts?: {
  targetAccuracy?: number;    // meters to stop early
  maxTime?: number;           // total time budget
  highAccTimeout?: number;    // time to wait before accepting coarse
  allowCachedMs?: number;     // accept cached fixes (max age)
}): Promise<Coords> {
  const {
    targetAccuracy = 25,
    maxTime = 12000,
    highAccTimeout = 6000,
    allowCachedMs = 120000, // 2 min
  } = opts || {};

  if (!("geolocation" in navigator)) {
    const last = loadLastKnown();
    if (last) return Promise.resolve(last);
    return Promise.reject(new Error("Geolocation unsupported"));
  }

  let best: GeolocationPosition | null = null;
  let resolved = false;
  let clearWatch = () => {};

  return new Promise<Coords>((resolve, reject) => {
    const finish = (pos: GeolocationPosition, source: string) => {
      if (resolved) return;
      resolved = true;
      clearWatch();
      const coords = { lat: pos.coords.latitude, lng: pos.coords.longitude, accuracy: pos.coords.accuracy, source };
      saveLastKnown(coords);
      resolve(coords);
    };

    // 1) High-accuracy watch (keeps improving)
    const watchId = navigator.geolocation.watchPosition(
      (pos) => {
        best = pos;
        if (pos.coords.accuracy && pos.coords.accuracy <= targetAccuracy) {
          finish(pos, "watchHighAcc");
        }
      },
      (_err) => {
        // ignore; we have coarse/lastKnown fallbacks
      },
      { enableHighAccuracy: true, maximumAge: 0, timeout: highAccTimeout }
    );
    clearWatch = () => navigator.geolocation.clearWatch(watchId);

    // 2) Quick coarse attempt (returns fast on many devices)
    navigator.geolocation.getCurrentPosition(
      (pos) => {
        if (!resolved && (!best || (pos.coords.accuracy < (best.coords.accuracy || Infinity)))) {
          best = pos;
        }
      },
      () => {},
      { enableHighAccuracy: false, maximumAge: allowCachedMs, timeout: 2500 }
    );

    // 3) High-accuracy “patience” timer → accept best-so-far or coarse
    const coarseTimer = setTimeout(() => {
      if (resolved) return;
      if (best) {
        finish(best, "bestSoFar");
      } else {
        const last = loadLastKnown(allowCachedMs);
        if (last) {
          resolved = true; clearWatch(); resolve(last);
        }
      }
    }, highAccTimeout);

    // 4) Hard deadline → return whatever we have or fail
    const hardTimer = setTimeout(() => {
      if (resolved) return;
      if (best) {
        finish(best, "deadlineBest");
      } else {
        const last = loadLastKnown(24 * 60 * 60 * 1000);
        clearWatch();
        if (last) resolve(last);
        else reject(new Error("Location timeout"));
      }
    }, maxTime);

    // safety: clear timers when resolved
    const stop = () => { clearTimeout(coarseTimer); clearTimeout(hardTimer); };
    const _origFinish = finish;
    // @ts-ignore
    finish = (...args) => { stop(); /* @ts-ignore */ _origFinish(...args); };
  });
}

2) Use it in your report flow (optimistic marker immediately)

Update your report handler (client/src/components/ReportIncidentModal.tsx or wherever you submit):

import { getAdaptivePosition, loadLastKnown } from "@/lib/location";

async function onSubmitIncident(form: { type: string; note?: string }) {
  // 1) show something immediately on the map
  const mapCluster = (window as any).__nnCluster;
  const addOptimistic = (lat: number, lng: number) => {
    if (!mapCluster) return null;
    const layer = L.circleMarker([lat, lng], { radius: 8, color: "#22c55e", weight: 2, fillOpacity: 0.9 })
      .addTo(mapCluster)
      .bindTooltip("Submitting…");
    (layer as any).__optimistic = true;
    return layer;
  };

  // 2) try adaptive location with graceful fallbacks
  let coords = loadLastKnown(5 * 60 * 1000); // recent last-known as instant seed
  let optimisticLayer: L.Layer | null = null;
  if (coords) optimisticLayer = addOptimistic(coords.lat, coords.lng);

  try {
    const fix = await getAdaptivePosition({ targetAccuracy: 25, maxTime: 12000, highAccTimeout: 6000 });
    coords = fix;
    if (!optimisticLayer) optimisticLayer = addOptimistic(fix.lat, fix.lng);
  } catch (e) {
    // still no coords? give a minimal UX choice or keep optimistic at last-known
    if (!coords) {
      // You can show a toast/confirm here. For now, bail with a friendly error:
      throw new Error("We couldn't get your GPS in time. Please ensure location is enabled and try again.");
    }
  }

  // 3) POST to server (works online; if offline your SW can queue)
  const payload = {
    type: form.type,
    note: form.note ?? "",
    latitude: coords!.lat,
    longitude: coords!.lng,
  };

  try {
    const res = await fetch("/api/incidents", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify(payload),
    });
    if (!res.ok) throw new Error(await res.text());

    const saved = await res.json();

    // Upgrade optimistic marker (optional: swap to the proper icon)
    if (optimisticLayer && (optimisticLayer as any).__optimistic) {
      (optimisticLayer as any).unbindTooltip?.();
      // replace with your custom icon marker for this type:
      // const icon = getIncidentIcon(form.type);
      // const m = L.marker([payload.latitude, payload.longitude], { icon }).addTo(mapCluster);
      // mapCluster.removeLayer(optimisticLayer);
    }
  } catch (err) {
    // If offline, rely on your "Saved offline" queue *but keep the optimistic marker visible*

  }
}

🟠 Replit → GPT (impact report)

This patch does not change your DB, sockets, or clustering architecture.

It replaces getAccuratePosition() with getAdaptivePosition():

resolves with best-so-far within 12s,

returns coarse or last-known if high-accuracy is slow,

caches the last known fix to keep the map responsive,

prevents the modal from failing with “Timeout expired.”

We add optimistic circle markers so the user always sees something, even offline.

Minimal UI changes; you can keep your existing icon system.
