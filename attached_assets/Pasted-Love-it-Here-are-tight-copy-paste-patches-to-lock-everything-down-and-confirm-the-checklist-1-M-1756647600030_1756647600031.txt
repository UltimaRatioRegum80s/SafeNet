  }) => {

    createIncidentMarker({
      id: inc.id,

      lat: inc.latitude,
      lng: inc.longitude,
      description: inc.description,
    });
  };

  socket.on("incident:new", onNew);
  return () => socket.off("incident:new", onNew);
}, []);

4) “Focus the new pin” support

Redirected with ?focus=<id>? Center when that marker exists.

Add a tiny util client/src/lib/url.ts:

export const getQueryParam = (key: string): string | null =>
  new URLSearchParams(window.location.search).get(key);


In Map.tsx after incidents load / marker hydration:

import { getQueryParam } from "@/lib/url";
import L from "leaflet";

useEffect(() => {
  const focusId = getQueryParam("focus");
  if (!focusId || !incidents?.length) return;

  // find the incident & center
  const target = incidents.find(i => i.id === focusId);
  if (!target) return;

  // center and flash
  map?.panTo([target.latitude, target.longitude], { animate: true });
  // optional pulse ring:
  L.circle([target.latitude, target.longitude], {
    radius: 20, color: "#22d3ee", fillColor: "#22d3ee", fillOpacity: 0.25
  }).addTo(map);

  // optional: cleanup querystring
  const u = new URL(window.location.href);
  u.searchParams.delete("focus");
  window.history.replaceState({}, "", u.toString());
}, [map, incidents]);


If you prefer optimistic focus (before refetch/socket), send coordinates in the redirect instead: ?lat=..&lng=.. and pan immediately.

5) QuickReportCard small fixes (useToast + race-proof)

client/src/features/report/QuickReportCard.tsx

import { useToast } from "@/components/ui/use-toast";
const { toast } = useToast();
const [isLocating, setIsLocating] = React.useState(false);
const isBusy = isLocating || createIncident.isPending;

const onSubmit = async () => {
  if (!type || !severity || isBusy) return;
  try {
    setIsLocating(true);
    const { lat, lng } = await getCurrentPosition();
    const result = await createIncident.mutateAsync({
      type, severity, latitude: lat, longitude: lng,
      description: note?.trim() || undefined,
    });

    reset();
    toast({ title: "Incident reported", description: "Redirecting to map…" });
    // Prefer focus param if server returns id
    const id = (result as any)?.id;
    setLocation(id ? `/map?focus=${encodeURIComponent(id)}` : "/map");
  } catch (err: any) {
    toast({
      title: "Couldn’t get your location",
      description: err?.message ?? "Please enable location and try again.",
      variant: "destructive",
    });
  } finally {
    setIsLocating(false);
  }
};


And disable the button:
