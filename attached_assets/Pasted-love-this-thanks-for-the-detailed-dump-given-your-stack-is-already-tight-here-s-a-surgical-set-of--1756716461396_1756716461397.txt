2) Self-healing & visibility (client-only, tiny)
2.1 Socket watchdog (no polling UX change)

You already have solid sockets; this fires a one-off fetch only if no socket traffic lands for a while.

// client/src/lib/socketWatchdog.ts
import { fetchIncidentsForBbox } from "../api/incidents"; // your existing API
import { getCurrentBbox } from "../features/map/bbox";
import { syncMarkers } from "../features/map/clusterSync";

let lastEventAt = Date.now();
export function hookSocketWatchdog(socket: Socket) {
  socket.onAny(() => (lastEventAt = Date.now())); // any event counts
  const t = setInterval(async () => {
    if (Date.now() - lastEventAt > 15000) {
      const bbox = getCurrentBbox();
      const fc = await fetchIncidentsForBbox(bbox, { sinceHours: 72 });
      syncMarkers(fc); // your “source of truth” reconcile
      lastEventAt = Date.now();
    }
  }, 5000);
  return () => clearInterval(t);
}


Call once when you init sockets on the map page.

2.2 Tiny status badge (dev-only)

Feature-flag a badge so @RDM can see socket/HTTP health in staging.

// client/src/components/NetBadge.tsx
export function NetBadge({ socket }: { socket: Socket }) {
  const [ws, setWs] = useState("⏳");
  const [http, setHttp] = useState("⏳");
  useEffect(() => {
    const onState = () => setWs(socket.connected ? "🟢" : "🟠");
    socket.on("connect", onState); socket.on("disconnect", onState);
    onState();
    const tick = setInterval(async () => {
      try {
        const r = await fetch("/api/healthz");
        setHttp(r.ok ? "🟢" : "🔴");
      } catch { setHttp("🔴"); }
    }, 8000);
    return () => { clearInterval(tick); socket.off("connect", onState); socket.off("disconnect", onState); };
  }, []);
  if (!import.meta.env.VITE_FLAG_NETBADGE) return null;
  return (
    <div className="fixed bottom-3 right-3 px-2 py-1 rounded bg-black/70 text-white text-xs">
      WS {ws} · HTTP {http}
    </div>
  );
}

5) Map layer hardening (Leaflet + MarkerCluster)
5.1 Single source of truth reconciliation

You already have seenIds. Add a periodic reconcile to remove ghosts (e.g., if a moderation event was missed).

// clusterSync.ts
export async function reconcileWithServer(bbox: Bbox) {
  const fc = await fetchIncidentsForBbox(bbox, { sinceHours: 72 });
  const serverIds = new Set(fc.features.map(f => f.properties.id));
  // remove markers not present server-side
  for (const id of localIds()) if (!serverIds.has(id)) removeMarker(id);
  // add/update present ones
  for (const f of fc.features) upsert(f.properties);
}


Call this from the watchdog or on tab visibility regain.

5.2 Invisible double-emit guard

If you sometimes receive both incident:new and incident:created:

socket.on("incident:new", (i) => upsertOnce(i));
socket.on("incident:created", (i) => upsertOnce(i));

const seen = new Set<string>();
function upsertOnce(i: Incident) {
  if (seen.has(i.id)) return;
  seen.add(i.id);
  upsertIncidentMarker(i);
}
