// NN Mobile Markers – Probe Pack v2 (read-only)
(() => {
  const out:any = {};
  const map = (window as any)._map || (window as any).map;
  const cluster = (window as any).clusterRef?.current || (window as any)._cluster;

  out.ua = navigator.userAgent;
  out.flagParam = new URL(location.href).searchParams.get('NN_FIX_MOBILE_MARKERS');
  out.flagLS = localStorage.getItem('NN_FIX_MOBILE_MARKERS');

  // Pane + DOM
  const pane = document.querySelector('.leaflet-marker-pane') as HTMLElement | null;
  out.markerPaneExists = !!pane;
  out.markerPaneDisplay = pane ? getComputedStyle(pane).display : 'none';
  out.markerPaneZ = pane ? getComputedStyle(pane).zIndex : '(none)';
  out.divIconWrappers = document.querySelectorAll('.nn-divicon-wrapper').length;
  out.clusterEls = document.querySelectorAll('.marker-cluster').length;

  // Cluster presence + contents
  out.hasMap = !!map;
  out.hasCluster = !!cluster;
  out.clusterOnMap = (map && cluster) ? map.hasLayer(cluster) : false;
  out.clusterLayerCount = cluster ? cluster.getLayers().length : 0;
  out.first5 = cluster ? cluster.getLayers().slice(0,5).map((l:any) => ({
    ctor: l?.constructor?.name,
    isMarker: l instanceof L.Marker,
    isCircle: l instanceof L.CircleMarker,
  })) : [];

  // MarkerCluster CSS?
  out.hasMCStyles = (() => {
    try {
      for (const s of Array.from(document.styleSheets)) {
        const rules = (s as CSSStyleSheet).cssRules || [];
        for (const r of Array.from(rules)) {
          if ((r as CSSStyleRule).selectorText?.includes('.marker-cluster')) return true;
        }
      }
    } catch (_) {} // cross-origin
    return false;
  })();

  console.table(out);

  // === Optional live probes (no code edits) ===
  // A) Add a single divIcon marker to the CLUSTER
  (out.doAddProbeToCluster = () => {
    if (!map || !cluster) return console.warn('no map/cluster');
    const c = map.getCenter();
    const icon = L.divIcon({ className:'nn-divicon-wrapper', html:'<div style="width:16px;height:16px;border-radius:9999px;background:rgba(37,99,235,.2);border:2px solid #2563eb"></div>', iconSize:[16,16], iconAnchor:[8,8] });
    const m = L.marker(c, { icon, title:'probe' });
    cluster.addLayer(m);
    console.log('added PROBE marker to cluster @', c, 'cluster layer count:', cluster.getLayers().length);
    return m;
  });

  // B) Add a single marker directly to the MAP (bypassing cluster)
  (out.doAddProbeDirectToMap = () => {
    if (!map) return console.warn('no map');
    const c = map.getCenter();
    const icon = L.divIcon({ className:'nn-divicon-wrapper', html:'<div style="width:16px;height:16px;border-radius:9999px;background:rgba(16,185,129,.2);border:2px solid #10b981"></div>', iconSize:[16,16], iconAnchor:[8,8] });
    const m = L.marker(c, { icon, title:'direct-probe' }).addTo(map);
    console.log('added DIRECT probe marker @', c);
    return m;
  });

  (window as any).__nnDiagOut = out;
  return out;
})();
