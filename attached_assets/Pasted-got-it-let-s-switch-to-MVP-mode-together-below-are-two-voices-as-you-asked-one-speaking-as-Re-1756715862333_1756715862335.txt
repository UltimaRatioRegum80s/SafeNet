Step 1 — Minimal Express API (in-memory)

Create (or replace) server/index.ts (or server.js if you’re not using TS). Keep it self-contained.

// server/index.ts
import express from "express";
import cors from "cors";

const app = express();
const PORT = process.env.PORT || 3001;

app.use(cors());
app.use(express.json());

// In-memory store (MVP mode)
type Incident = {
  id: string;
  lat: number; // latitude
  lng: number; // longitude
  level: "yellow" | "amber" | "red";
  createdAt: number;
};

const incidents: Incident[] = [];

// POST /api/incidents  -> add one
app.post("/api/incidents", (req, res) => {
  const { lat, lng, level } = req.body ?? {};
  const validLevel = level === "yellow" || level === "amber" || level === "red";

  if (
    typeof lat !== "number" ||
    typeof lng !== "number" ||
    !validLevel ||
    Number.isNaN(lat) ||
    Number.isNaN(lng)
  ) {
    return res.status(400).json({ ok: false, error: "Invalid payload" });
  }

  const inc: Incident = {
    id: `${Date.now()}-${Math.random().toString(36).slice(2, 7)}`,
    lat,
    lng,
    level,
    createdAt: Date.now(),
  };
  incidents.push(inc);
  return res.json({ ok: true, incident: inc });
});

// GET /api/incidents  -> GeoJSON FeatureCollection
app.get("/api/incidents", (_req, res) => {
  const featureCollection = {
    type: "FeatureCollection",
    features: incidents.map((i) => ({
      type: "Feature",
      properties: { id: i.id, level: i.level, createdAt: i.createdAt },
      geometry: { type: "Point", coordinates: [i.lng, i.lat] }, // [lng, lat] per GeoJSON spec
    })),
  };
  res.json(featureCollection);
});

app.listen(PORT, () => {
  console.log(`MVP incidents API running on :${PORT}`);
});


Replit note: ensure the server run command exposes this port; if your frontend runs on a separate port, set VITE_API_URL (see next step) to your Replit server URL.

Step 2 — Tiny API client

Create src/lib/api.ts:

// src/lib/api.ts
export type Level = "yellow" | "amber" | "red";

const API_URL = import.meta.env.VITE_API_URL || "http://localhost:3001";

export async function postIncident(input: {
  lat: number;
  lng: number;
  level: Level;
}) {
  const res = await fetch(`${API_URL}/api/incidents`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(input),
  });
  if (!res.ok) throw new Error(await res.text());
  return res.json();
}

export async function fetchIncidentsGeoJSON(): Promise<GeoJSON.FeatureCollection> {
  const res = await fetch(`${API_URL}/api/incidents`);
  if (!res.ok) throw new Error(await res.text());
  return res.json();
}

Step 3 — Report page: one button + level picker

Create/replace src/pages/ReportPage.tsx:

// src/pages/ReportPage.tsx
import { useState } from "react";
import { postIncident, type Level } from "../lib/api";

export default function ReportPage() {
  const [level, setLevel] = useState<Level>("yellow");
  const [busy, setBusy] = useState(false);
  const [msg, setMsg] = useState<string | null>(null);

  const handleReport = async () => {
    setMsg(null);
    setBusy(true);
    try {
      const coords = await getCoords();
      await postIncident({ lat: coords.lat, lng: coords.lng, level });
      setMsg("Incident reported. Check the map.");
    } catch (e: any) {
      setMsg(e?.message || "Failed to report");
    } finally {
      setBusy(false);
    }
  };

  return (
    <div className="mx-auto max-w-md p-4 space-y-4">
      <h1 className="text-xl font-semibold">Report Incident</h1>

      <div className="flex gap-2">
        <LevelButton current={level} value="yellow" onClick={setLevel} />
        <LevelButton current={level} value="amber" onClick={setLevel} />
        <LevelButton current={level} value="red" onClick={setLevel} />
      </div>

      <button
        onClick={handleReport}
        disabled={busy}
        className="px-4 py-2 rounded-2xl bg-blue-600 text-white disabled:opacity-50 shadow"
      >
        {busy ? "Reporting..." : "Report Incident Here"}
      </button>

      {msg && <p className="text-sm text-gray-700">{msg}</p>}
    </div>
  );
}

function LevelButton({
  current,
  value,
  onClick,
}: {
  current: Level;
  value: Level;
  onClick: (l: Level) => void;
}) {
  const colors: Record<Level, string> = {
    yellow: "bg-yellow-400",
    amber: "bg-amber-500",
    red: "bg-red-500",
  };
  return (
    <button
      onClick={() => onClick(value)}
      className={`px-3 py-1 rounded-2xl text-white ${colors[value]} ${
        current === value ? "ring-2 ring-black/30" : ""
      }`}
    >
      {value}
    </button>
  );
}

function getCoords(): Promise<{ lat: number; lng: number }> {
  return new Promise((resolve, reject) => {
    if (!navigator.geolocation) {
      return reject(new Error("Geolocation not supported"));
    }
    navigator.geolocation.getCurrentPosition(
      (pos) => {
        const { latitude, longitude } = pos.coords;
        resolve({ lat: latitude, lng: longitude });
      },
      (err) => reject(new Error(err.message || "Geolocation failed")),
      { enableHighAccuracy: true, timeout: 10000, maximumAge: 0 }
    );
  });
}

Step 4 — Map page: poll & render circles (no UI changes)

Replace the data plumbing only; keep your existing map layout/controls. Example src/pages/MapPage.tsx:

// src/pages/MapPage.tsx
import { useEffect, useRef } from "react";
import maplibregl, { Map, GeoJSONSource } from "maplibre-gl";
import "maplibre-gl/dist/maplibre-gl.css";
import { useQuery } from "@tanstack/react-query";
import { fetchIncidentsGeoJSON } from "../lib/api";

export default function MapPage() {
  const mapRef = useRef<Map | null>(null);
  const containerRef = useRef<HTMLDivElement | null>(null);

  // Poll incidents every 2s
  const { data } = useQuery({
    queryKey: ["incidents"],
    queryFn: fetchIncidentsGeoJSON,
    refetchInterval: 2000,
  });

  useEffect(() => {
    if (!containerRef.current || mapRef.current) return;

    const map = new maplibregl.Map({
      container: containerRef.current,
      style: "https://demotiles.maplibre.org/style.json",
      center: [18.4241, -33.9249], // [lng, lat] — set to your area
      zoom: 12,
    });
    mapRef.current = map;

    map.on("load", () => {
      // initial empty source
      map.addSource("incidents", {
        type: "geojson",
        data: { type: "FeatureCollection", features: [] },
      });

      map.addLayer({
        id: "incidents-circles",
        type: "circle",
        source: "incidents",
        paint: {
          "circle-radius": 8,
          "circle-stroke-width": 1.5,
          "circle-stroke-color": "#ffffff",
          "circle-color": [
            "match",
            ["get", "level"],
            "yellow",
            "#facc15",
            "amber",
            "#f59e0b",
            "red",
            "#ef4444",
            "#9ca3af",
          ],
        },
      });
    });

    return () => map.remove();
  }, []);

  // Push new data into the source whenever it changes
  useEffect(() => {
    if (!data || !mapRef.current) return;
    const src = mapRef.current.getSource("incidents") as GeoJSONSource | undefined;
    if (src) src.setData(data);
  }, [data]);

  return <div ref={containerRef} className="w-full h-[calc(100vh-80px)]" />;
}


Key detail: GeoJSON coordinates order is [lng, lat]. Our server already outputs that; don’t flip them on the client.
