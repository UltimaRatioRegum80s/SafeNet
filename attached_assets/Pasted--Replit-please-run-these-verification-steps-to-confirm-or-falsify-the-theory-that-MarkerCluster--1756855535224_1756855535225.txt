2) One-paste CSS presence check (copy/paste)
// NN MarkerCluster CSS Verify (read-only)
(() => {
  const out = {};
  out.clusterLayers = window._cluster?.getLayers().length ?? 'no _cluster';
  out.clusterEls = document.querySelectorAll('.marker-cluster').length;

  // find any loaded rule that targets .marker-cluster
  out.hasMCStyles = false;
  out.sampleRule = null;
  try {
    for (const s of Array.from(document.styleSheets)) {
      const rules = s.cssRules || [];
      for (const r of Array.from(rules)) {
        const sel = (r.selectorText || '');
        if (sel.includes('.marker-cluster')) {
          out.hasMCStyles = true;
          out.sampleRule = r.cssText.slice(0, 160);
          throw 'found';
        }
      }
    }
  } catch(_) {}

  const el = document.querySelector('.marker-cluster');
  out.computed = el ? {
    width:  getComputedStyle(el).width,
    height: getComputedStyle(el).height,
    display:getComputedStyle(el).display
  } : null;

  console.table(out);
  return out;
})();
