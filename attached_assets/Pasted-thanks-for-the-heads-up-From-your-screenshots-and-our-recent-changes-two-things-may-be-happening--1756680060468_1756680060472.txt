1) Ensure we always send good coordinates from the form
// client/src/features/report/useMobileLocation.ts
export async function getAccuratePosition(): Promise<{lat:number;lng:number}> {
  return new Promise((resolve, reject) => {
    if (!navigator.geolocation) return reject(new Error("Geolocation unavailable"));
    navigator.geolocation.getCurrentPosition(
      (pos) => {
        const lat = Number(pos.coords.latitude.toFixed(6));
        const lng = Number(pos.coords.longitude.toFixed(6));
        resolve({ lat, lng });
      },
      (err) => reject(err),
      { enableHighAccuracy: true, timeout: 12000, maximumAge: 0 }
    );
  });
}


Use it in your incident form right before submit:

// client/src/features/report/submitIncident.ts
import { getAccuratePosition } from "./useMobileLocation";
import { RoutePaths } from "@/router/routePaths";
import { useCreateIncident } from "@/hooks/useCreateIncident";
import { useNavigate } from "react-router-dom";

export function useSubmitIncident() {
  const mutate = useCreateIncident();
  const navigate = useNavigate();

  return async (values: any) => {
    let coords = values; // might already have lat/lng from a map picker
    if (!Number.isFinite(+values.lat) || !Number.isFinite(+values.lng)) {
      try {
        const p = await getAccuratePosition();
        coords = { ...values, lat: p.lat, lng: p.lng };
      } catch {
        // fall back to a friendly error or allow submit without map jump
      }
    }

    // Store last incident locally so map can highlight even if socket is slow
    const last = { lat: +coords.lat, lng: +coords.lng, type: values.type, t: Date.now() };
    localStorage.setItem("nn:last-incident", JSON.stringify(last));

    // submit (optimistic handled by hook)
    await mutate.mutateAsync(coords);

    // Redirect WITH focus params so the map pans/zooms to your spot
    const p = new URLSearchParams({
      lat: String(last.lat),
      lng: String(last.lng),
      z: "17",
      highlight: "1",
    });
    navigate(`${RoutePaths.Map}?${p.toString()}`, { replace: true });
  };
}

2) Make the map honor focus params and visibly highlight the new marker
// client/src/components/InteractiveMap.tsx
import { useEffect, useMemo, useRef } from "react";
import { useLocation } from "react-router-dom";
// ...existing imports...

function useFocusParams() {
  const { search } = useLocation();
  const sp = useMemo(() => new URLSearchParams(search), [search]);
  const lat = Number(sp.get("lat"));
  const lng = Number(sp.get("lng"));
  const z = Number(sp.get("z") ?? 17);
  const highlight = sp.get("highlight") === "1";
  return { lat, lng, z, highlight, valid: Number.isFinite(lat) && Number.isFinite(lng) };
}

export default function InteractiveMap() {
  // ...existing refs and queries...
  const focus = useFocusParams();

  // After map init & markers render, pan/zoom to focused point and drop a pulse
  useEffect(() => {
    if (!mapRef.current || !clusterRef.current) return;

    const map = mapRef.current;
    if (focus.valid) {
      map.setView([focus.lat, focus.lng], focus.z, { animate: true });

      if (focus.highlight) {
        const pulse = L.circleMarker([focus.lat, focus.lng], {
          radius: 16,
          color: "#ef4444",
          fillColor: "#ef4444",
          fillOpacity: 0.25,
          weight: 2,
          className: "nn-pulse",
        }).addTo(map);

        // auto-remove after a few seconds
        setTimeout(() => {
          try { map.removeLayer(pulse); } catch {}
        }, 4000);
      }
    }
  }, [focus.valid, focus.lat, focus.lng, focus.z, focus.highlight]);

3) Ensure optimistic marker uses the same normalized coords (no “invisible” optimistic)

In your useCreateIncident hook, normalize before seeding cache:

// client/src/hooks/useCreateIncident.ts
const norm = (v:any) => ({
  ...v,
  lat: Number.parseFloat(v.lat as any),
  lng: Number.parseFloat(v.lng as any),
});

onMutate: async (vars: any) => {
  await qc.cancelQueries({ queryKey: ["incidents"] });
  const previous = qc.getQueryData<any[]>(["incidents"]) || [];
  const nv = norm(vars);
  if (!Number.isFinite(nv.lat) || !Number.isFinite(nv.lng)) {
    // don't seed an optimistic marker with bad coords
    return { previous };
  }
  const optimistic = { ...nv, id: `temp-${Date.now()}`, __optimistic: true };
  qc.setQueryData(["incidents"], [optimistic, ...previous]);
  // also stash for map focus as a backup signal
  localStorage.setItem("nn:last-incident", JSON.stringify({ lat:nv.lat, lng:nv.lng, t:Date.now() }));
  return { previous };
},

4) Map fallback: if URL params are missing, use the locally stored last incident once
// in InteractiveMap.tsx, after focus effect
useEffect(() => {
  if (!mapRef.current || focus.valid) return;

  const raw = localStorage.getItem("nn:last-incident");
  if (!raw) return;
  try {
    const { lat, lng, t } = JSON.parse(raw);
    if (Date.now() - t < 5 * 60 * 1000 && Number.isFinite(lat) && Number.isFinite(lng)) {
      mapRef.current!.setView([lat, lng], 17, { animate: true });
      // don’t spam; clear after use
      localStorage.removeItem("nn:last-incident");
    }
  } catch {}
}, [focus.valid]);
