# NN GPS & Location Services

A production-ready implementation for NaborNet’s PWA that provides:

* **Accurate GPS with network/cell fallback** (via Geolocation API first; graceful fallback to IP-based geolocation through a backend endpoint).
* **User opt-in + clear permission rationale UI** (shadcn/ui component).
* **Throttled foreground updates** (watchPosition) and **best-effort background polling** (when tab/app is backgrounded, within browser/OS limits).
* **State management with Zustand** and persistence of last known location.

> ⚠️ Note on the Web Platform: Service Workers **cannot** access geolocation and true continuous background GPS is not available in browsers. We implement: (1) real-time `watchPosition` while the app is foregrounded/active; (2) reduced-frequency polling while hidden/installed PWA is backgrounded, which the OS may throttle. We surface clear UX around this.

## 1) Types & Utilities

**`src/lib/location/types.ts`**

```ts
export type LocationSource = 'gps-high' | 'gps-network' | 'ip-fallback';

export type PermissionStateLike = 'granted' | 'denied' | 'prompt' | 'unsupported';

export interface NNPosition {
  lat: number;
  lon: number;
  accuracy: number; // meters (lower is better)
  altitude?: number | null;
  altitudeAccuracy?: number | null;
  heading?: number | null;
  speed?: number | null; // m/s
  timestamp: number; // epoch ms
  source: LocationSource;
}

export interface WatchOptions {
  /** High-accuracy GPS first (true), else network-based first (false). */
  highAccuracy?: boolean;
  /** Emit only if moved this much since last emit (meters). */
  minDistanceM?: number; // e.g. 25m
  /** Emit at most once per this interval (ms). */
  minIntervalMs?: number; // e.g. 15_000
  /** When app hidden, poll no more than once per this interval (ms). */
  backgroundIntervalMs?: number; // e.g. 180_000 (3m)
  /** Timeout for high-accuracy attempt (ms). */
  highAccTimeoutMs?: number; // e.g. 8_000
  /** Timeout for low-accuracy attempt (ms). */
  lowAccTimeoutMs?: number; // e.g. 8_000
}
```

**`src/lib/location/utils.ts`**

```ts
export function haversineMeters(
  a: { lat: number; lon: number },
  b: { lat: number; lon: number }
): number {
  const R = 6371_000; // meters
  const toRad = (d: number) => (d * Math.PI) / 180;
  const dLat = toRad(b.lat - a.lat);
  const dLon = toRad(b.lon - a.lon);
  const lat1 = toRad(a.lat);
  const lat2 = toRad(b.lat);

  const sinDLat = Math.sin(dLat / 2);
  const sinDLon = Math.sin(dLon / 2);

  const h = sinDLat * sinDLat + Math.cos(lat1) * Math.cos(lat2) * sinDLon * sinDLon;
  const c = 2 * Math.atan2(Math.sqrt(h), Math.sqrt(1 - h));
  return R * c;
}

export function withTimeout<T>(promise: Promise<T>, ms: number, label = 'timeout'): Promise<T> {
  return new Promise((resolve, reject) => {
    const t = setTimeout(() => reject(new Error(label)), ms);
    promise
      .then((v) => {
        clearTimeout(t);
        resolve(v);
      })
      .catch((e) => {
        clearTimeout(t);
        reject(e);
      });
  });
}

export function isSecureContextOrLocalhost(): boolean {
  if (typeof window === 'undefined') return false;
  if (window.isSecureContext) return true;
  const host = window.location.hostname;
  return host === 'localhost' || host === '127.0.0.1';
}
```

## 3) Geolocation Service (with network/IP fallback)

**`src/lib/location/service.ts`**

```ts
import type { NNPosition, WatchOptions, PermissionStateLike, LocationSource } from './types';
import { haversineMeters, withTimeout, isSecureContextOrLocalhost } from './utils';

const DEFAULTS: Required<WatchOptions> = {
  highAccuracy: true,
  minDistanceM: 25,
  minIntervalMs: 15_000,
  backgroundIntervalMs: 180_000, // 3 minutes
  highAccTimeoutMs: 8_000,
  lowAccTimeoutMs: 8_000,
};

export async function queryPermission(): Promise<PermissionStateLike> {
  if (typeof navigator === 'undefined' || !('geolocation' in navigator)) return 'unsupported';
  try {
    // @ts-ignore - TS doesn't know the union here
    const status = await navigator.permissions?.query({ name: 'geolocation' as PermissionName });
    return (status?.state as PermissionStateLike) ?? 'prompt';
  } catch {
    // Some browsers throw; treat as prompt
    return 'prompt';
  }
}

function toNNPosition(pos: GeolocationPosition, fromHighAcc: boolean): NNPosition {
  const c = pos.coords;
  const src: LocationSource = fromHighAcc ? 'gps-high' : 'gps-network';
  return {
    lat: c.latitude,
    lon: c.longitude,
    accuracy: c.accuracy,
    altitude: c.altitude ?? null,
    altitudeAccuracy: c.altitudeAccuracy ?? null,
    heading: c.heading ?? null,
    speed: c.speed ?? null,
    timestamp: pos.timestamp,
    source: src,
  };
}

async function getViaGeolocation(enableHighAccuracy: boolean, timeoutMs: number): Promise<NNPosition> {
  if (typeof navigator === 'undefined' || !navigator.geolocation) throw new Error('Geolocation unsupported');
  return withTimeout(
    new Promise<GeolocationPosition>((resolve, reject) => {
      navigator.geolocation.getCurrentPosition(resolve, reject, {
        enableHighAccuracy,
        maximumAge: 0,
        timeout: timeoutMs,
      });
    }),
    timeoutMs + 2000,
    'geolocation-timeout'
  ).then((p) => toNNPosition(p, enableHighAccuracy));
}

export async function getIpFallback(): Promise<NNPosition> {
  const res = await fetch('/api/geo/ip', { headers: { 'x-nn-client': 'web' } });
  if (!res.ok) throw new Error('ip-fallback-failed');
  const j = await res.json();
  const now = Date.now();
  return {
    lat: j.lat,
    lon: j.lon,
    accuracy: j.accuracy ?? 25_000, // IP geo is coarse (~10-25km)
    altitude: null,
    altitudeAccuracy: null,
    heading: null,
    speed: null,
    timestamp: now,
    source: 'ip-fallback',
  } satisfies NNPosition;
}

/**
 * Try high-accuracy first, then low-accuracy, then IP as last resort.
 */
export async function getCurrentPositionSmart(opts?: Partial<WatchOptions>): Promise<NNPosition> {
  const o = { ...DEFAULTS, ...opts };
  if (!isSecureContextOrLocalhost()) throw new Error('Geolocation requires HTTPS or localhost');
  try {
    return await getViaGeolocation(true, o.highAccTimeoutMs);
  } catch (e1) {
    try {
      return await getViaGeolocation(false, o.lowAccTimeoutMs);
    } catch (e2) {
      return await getIpFallback();
    }
  }
}

export interface WatchHandle { stop: () => void; }

export function startLocationWatch(
  onUpdate: (pos: NNPosition) => void,
  onError: (err: GeolocationPositionError | Error) => void,
  options?: Partial<WatchOptions>
): WatchHandle {
  const o = { ...DEFAULTS, ...options };
  if (!isSecureContextOrLocalhost()) {
    // Immediately fall back to IP if not secure context
    getIpFallback().then(onUpdate).catch(onError);
    return { stop: () => void 0 };
  }

  let lastEmitted: NNPosition | undefined;
  let lastEmitTs = 0;
  let watchId: number | null = null;
  let bgTimer: number | null = null;

  const clearWatch = () => {
    if (watchId != null) {
      navigator.geolocation.clearWatch(watchId);
      watchId = null;
    }
  };
  const clearBg = () => {
    if (bgTimer != null) {
      clearInterval(bgTimer);
      bgTimer = null;
    }
  };

  const shouldEmit = (p: NNPosition) => {
    const now = Date.now();
    const intervalOk = now - lastEmitTs >= o.minIntervalMs;
    const distanceOk = !lastEmitted || haversineMeters(lastEmitted, p) >= o.minDistanceM;
    return intervalOk && distanceOk;
  };

  const handleSuccess = (pos: GeolocationPosition, fromHighAcc: boolean) => {
    const nn = toNNPosition(pos, fromHighAcc);
    if (shouldEmit(nn)) {
      lastEmitted = nn;
      lastEmitTs = Date.now();
      onUpdate(nn);
    }
  };

  const startForegroundWatch = () => {
    clearWatch();
    watchId = navigator.geolocation.watchPosition(
      (p) => handleSuccess(p, o.highAccuracy ?? true),
      async (err) => {
        // Fallback: try low-accuracy once, then IP
        if (o.highAccuracy) {
          try {
            const low = await getViaGeolocation(false, o.lowAccTimeoutMs);
            if (shouldEmit(low)) {
              lastEmitted = low;
              lastEmitTs = Date.now();
              onUpdate(low);
            }
            return;
          } catch (_) {}
        }
        try {
          const ip = await getIpFallback();
          if (shouldEmit(ip)) {
            lastEmitted = ip;
            lastEmitTs = Date.now();
            onUpdate(ip);
          }
        } catch (e) {
          onError(err);
        }
      },
      {
        enableHighAccuracy: o.highAccuracy,
        maximumAge: 2_000,
        timeout: o.highAccTimeoutMs,
      }
    );
  };

  const startBackgroundPoll = () => {
    clearBg();
    bgTimer = window.setInterval(async () => {
      try {
        const pos = await getCurrentPositionSmart({
          highAccTimeoutMs: 4_000,
          lowAccTimeoutMs: 4_000,
        });
        if (shouldEmit(pos)) {
          lastEmitted = pos;
          lastEmitTs = Date.now();
          onUpdate(pos);
        }
      } catch (e) {
        // swallow; next tick will try again
      }
    }, o.backgroundIntervalMs);
  };

  // Initial active watch
  startForegroundWatch();

  const onVisChange = () => {
    const hidden = document.visibilityState === 'hidden';
    if (hidden) {
      clearWatch();
      startBackgroundPoll();
    } else {
      clearBg();
      startForegroundWatch();
    }
  };
  document.addEventListener('visibilitychange', onVisChange);

  return {
    stop: () => {
      clearWatch();
      clearBg();
      document.removeEventListener('visibilitychange', onVisChange);
    },
  };
}
```

## 4) React Hook + Integration

**`src/hooks/useLocation.ts`**

```ts
import { useEffect, useRef } from 'react';
import { useLocationStore } from '@/store/locationStore';
import { queryPermission, startLocationWatch, getCurrentPositionSmart } from '@/lib/location/service';
import type { WatchHandle } from '@/lib/location/service';

export function useBootstrapLocation() {
  const setPermission = useLocationStore((s) => s.setPermission);
  const setLast = useLocationStore((s) => s.setLast);
  const setError = useLocationStore((s) => s.setError);

  useEffect(() => {
    let cancelled = false;
    (async () => {
      const p = await queryPermission();
      if (!cancelled) setPermission(p);
      if (p === 'granted') {
        try {
          const pos = await getCurrentPositionSmart();
          if (!cancelled) setLast(pos);
        } catch (e: any) {
          if (!cancelled) setError(e?.message ?? 'location-init-failed');
        }
      }
    })();
    return () => {
      cancelled = true;
    };
  }, [setPermission, setLast, setError]);
}

export function useLiveLocation() {
  const setLast = useLocationStore((s) => s.setLast);
  const setWatching = useLocationStore((s) => s.setWatching);
  const setError = useLocationStore((s) => s.setError);
  const handleRef = useRef<WatchHandle | null>(null);

  useEffect(() => {
    setWatching(true);
    handleRef.current = startLocationWatch(
      (pos) => setLast(pos),
      (err) => setError(err?.message ?? 'watch-error'),
      {
        highAccuracy: true,
        minDistanceM: 25,
        minIntervalMs: 15_000,
        backgroundIntervalMs: 180_000,
      }
    );
    return () => {
      setWatching(false);
      handleRef.current?.stop();
      handleRef.current = null;
    };
  }, [setLast, setWatching, setError]);
}
```

## 6) Backend: IP-based Fallback + Telemetry Endpoint (Express)

**`server/routes/geo.ts`**

```ts
import express from 'express';
import fetch from 'node-fetch';

const router = express.Router();

/**
 * GET /api/geo/ip
 * Uses a provider to estimate coarse location from IP as a last-resort fallback.
 * Configure one of: IPAPI_CO_TOKEN, IPINFO_TOKEN, or a custom URL.
 */
router.get('/ip', async (req, res) => {
  try {
    const ip =
      (req.headers['x-forwarded-for'] as string)?.split(',')[0]?.trim() ||
      (req.socket as any)?.remoteAddress ||
      undefined;

    const provider = process.env.GEO_FALLBACK_PROVIDER || 'ipapi'; // 'ipinfo' | 'ipdata' | 'custom'
    let lat = 0,
      lon = 0,
      accuracy = 25_000;

    if (provider === 'ipinfo') {
      const token = process.env.IPINFO_TOKEN;
      const url = `https://ipinfo.io/${ip ? ip : ''}?token=${token}`;
      const r = await fetch(url);
      if (!r.ok) throw new Error('ipinfo-failed');
      const j = await r.json();
      if (j.loc) {
        const [la, lo] = j.loc.split(',').map(Number);
        lat = la;
        lon = lo;
      }
    } else if (provider === 'ipapi') {
      // ipapi.co works without a token but has rate limits; use token if available
      const url = ip
        ? `https://ipapi.co/${ip}/json/`
        : `https://ipapi.co/json/`;
      const r = await fetch(url);
      if (!r.ok) throw new Error('ipapi-failed');
      const j = await r.json();
      lat = Number(j.latitude);
      lon = Number(j.longitude);
      accuracy = 15_000;
    } else if (provider === 'custom' && process.env.IP_GEO_API_URL) {
      const url = process.env.IP_GEO_API_URL!;
      const r = await fetch(url);
      if (!r.ok) throw new Error('custom-geo-failed');
      const j = await r.json();
      lat = Number(j.lat ?? j.latitude);
      lon = Number(j.lon ?? j.longitude);
      accuracy = Number(j.accuracy ?? 25_000);
    } else {
      throw new Error('no-fallback-provider-configured');
    }

    res.json({ lat, lon, accuracy, ip: ip ?? null });
  } catch (e: any) {
    res.status(500).json({ error: e?.message ?? 'ip-geo-error' });
  }
});

/**
 * POST /api/geo/telemetry
 * Example endpoint that accepts client-side location updates.
 */
router.post('/telemetry', express.json(), async (req, res) => {
  try {
    const body = req.body;
    // TODO: optionally validate & store to Postgres via Drizzle
    // e.g., queue write for batching to reduce DB load
    res.json({ ok: true });
  } catch (e: any) {
    res.status(500).json({ error: e?.message ?? 'telemetry-error' });
  }
});

export default router;
```

## 7) Wiring & Defaults

* Add the **store, utils, and hook** files under `src/` as shown.
* Render `<LocationConsentCard />` somewhere prominent (e.g., onboarding or settings panel). The live hook can be activated at app root if you want constant updates while the app is open.
* Ensure your dev/preview runs on **HTTPS** (PWA requirement for geolocation) or `localhost`.
* Configure one fallback provider via environment:

  * `GEO_FALLBACK_PROVIDER=ipapi` *(default, low-friction, coarse)*
  * Or `ipinfo` with `IPINFO_TOKEN=...`
  * Or `custom` with `IP_GEO_API_URL=https://...` that returns `{ lat, lon, accuracy }`.

---

## 8) Acceptance Mapping

* **Accurate GPS + cell-tower/network fallback**: high-accuracy first → low-accuracy → IP fallback.
* **User opt-in**: `LocationConsentCard` prompts and explains rationale.
* **Background updates guarded by OS limits**: foreground `watchPosition` + hidden-tab polling with conservative interval; OS may clamp further.

---

## 10) Wiring into existing `ReportIncident.tsx`

**`client/src/pages/ReportIncident.tsx` (drop-in example)**

```tsx
import React, { useEffect, useMemo, useRef, useState } from 'react';
import { GeolocationService, type LocationData } from '@/lib/geolocationService';
import { Button } from '@/components/ui/button';
import { Alert, AlertDescription } from '@/components/ui/alert';

const geo = new GeolocationService();

const WINDHOEK = { lat: -22.5597, lon: 17.0658 };

export default function ReportIncident() {
  const [loc, setLoc] = useState<LocationData | null>(null);
  const [locError, setLocError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);
  const [accuracyThreshold, setAccuracyThreshold] = useState(100); // meters
  const stopRef = useRef<() => void>();

  // bootstrap location once on mount (prompt + best effort)
  useEffect(() => {
    let cancelled = false;
    (async () => {
      try {
        const p = await geo.getCurrentPosition({
          highAccTimeoutMs: 8000,
          lowAccTimeoutMs: 8000,
        });
        if (!cancelled) setLoc(p);
      } catch (e: any) {
        if (!cancelled) setLocError(e?.message ?? 'location-init-failed');
      }
    })();
    return () => {
      cancelled = true;
      stopRef.current?.();
    };
  }, []);

  // live updates while the form is open
  const startLive = async () => {
    setLocError(null);
    geo.startWatchingPosition(
      (p) => setLoc(p),
      (err) => setLocError(err?.message ?? 'watch-error'),
      {
        highAccuracy: true,
        minDistanceM: 25,
        minIntervalMs: 15_000,
        backgroundIntervalMs: 180_000,
      }
    );
    stopRef.current = () => geo.stopWatchingPosition();
  };

  const stopLive = () => {
    geo.stopWatchingPosition();
    stopRef.current = undefined;
  };

  const finalLatLon = useMemo(() => {
    if (!loc) return { lat: WINDHOEK.lat, lon: WINDHOEK.lon, isDefault: true };
    if (loc.accuracy > accuracyThreshold) {
      return { lat: WINDHOEK.lat, lon: WINDHOEK.lon, isDefault: true };
    }
    return { lat: loc.latitude, lon: loc.longitude, isDefault: false };
  }, [loc, accuracyThreshold]);

  async function submitIncident(e: React.FormEvent) {
    e.preventDefault();
    setLoading(true);
    try {
      // Example payload; add your other form fields
      const payload = {
        type: 'incident',
        latitude: finalLatLon.lat,
        longitude: finalLatLon.lon,
        accuracy: loc?.accuracy ?? null,
        locSource: loc?.source ?? 'unknown',
        // ... other form fields
      };

      // Send telemetry update for the user (optional):
      try {
        await fetch(`/api/users/me/location`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ latitude: payload.latitude, longitude: payload.longitude }),
        });
      } catch {}

      // Create the incident
      const res = await fetch('/api/incidents', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(payload),
      });
      if (!res.ok) throw new Error('failed-to-create-incident');

      // Optionally, check for duplicates
      try {
        await fetch(`/api/incidents/duplicates`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({
            type: 'incident',
            latitude: payload.latitude,
            longitude: payload.longitude,
          }),
        });
      } catch {}

      // Navigate away or show success
      setLoading(false);
      alert('Incident reported successfully');
    } catch (e: any) {
      setLoading(false);
      setLocError(e?.message ?? 'submit-error');
    }
  }

  return (
    <div className="mx-auto max-w-xl space-y-4 p-4">
      <div className="flex items-center justify-between gap-2">
        <h1 className="text-lg font-semibold">Report Incident</h1>
        <div className="text-xs text-muted-foreground">
          {loc ? `±${Math.round(loc.accuracy)} m • ${loc.source}` : 'locating…'}
        </div>
      </div>

      {locError && (
        <Alert variant="destructive"><AlertDescription className="text-xs">{locError}</AlertDescription></Alert>
      )}

      <div className="flex gap-2">
        <Button size="sm" onClick={startLive}>Improve Accuracy</Button>
        <Button size="sm" variant="secondary" onClick={stopLive}>Stop</Button>
      </div>

      {loc && loc.accuracy > accuracyThreshold && (
        <Alert>
          <AlertDescription className="text-xs">
            Location accuracy too low (±{Math.round(loc.accuracy)} m). Using default Windhoek until signal improves.
          </AlertDescription>
        </Alert>
      )}

      {/* Your form fields here */}
      <form onSubmit={submitIncident} className="space-y-3">
        {/* ... */}
        <Button type="submit" disabled={loading} className="w-full">
          {loading ? 'Submitting…' : 'Submit incident'}
        </Button>
      </form>

      <div className="text-xs text-muted-foreground">
        Using: {finalLatLon.isDefault ? 'Windhoek default' : 'Current position'}
      </div>
    </div>
  );
}
```

```ts
import express from 'express';
import { Pool } from 'pg';

const router = express.Router();
const pool = new Pool({ connectionString: process.env.DATABASE_URL });

// Update current user location (assumes auth middleware sets req.user.id)
router.post('/users/me/location', express.json(), async (req, res) => {
  try {
    const userId = (req as any).user?.id || null; // replace with your auth
    if (!userId) return res.status(401).json({ error: 'unauthorized' });
    const { latitude, longitude } = req.body;
    if (typeof latitude !== 'number' || typeof longitude !== 'number') {
      return res.status(400).json({ error: 'invalid-coordinates' });
    }

    await pool.query(
      `UPDATE users
       SET lat = $1, lon = $2, updated_at = NOW()
       WHERE id = $3`,
      [latitude, longitude, userId]
    );
    res.json({ ok: true });
  } catch (e: any) {
    res.status(500).json({ error: e?.message ?? 'update-location-error' });
  }
});

// Nearby Incidents
router.get('/incidents/nearby', async (req, res) => {
  try {
    const lat = Number(req.query.lat);
    const lon = Number(req.query.lon);
    const radius = Number(req.query.radius ?? 2000);
    if (Number.isNaN(lat) || Number.isNaN(lon)) return res.status(400).json({ error: 'invalid-coordinates' });

    const { rows } = await pool.query(
      `SELECT id, title, lat, lon,
              earth_distance(ll_to_earth($1,$2), ll_to_earth(lat, lon)) AS meters
       FROM incidents
       WHERE earth_distance(ll_to_earth($1,$2), ll_to_earth(lat, lon)) <= $3
       ORDER BY meters ASC
       LIMIT 100`,
      [lat, lon, radius]
    );
    res.json(rows);
  } catch (e: any) {
    res.status(500).json({ error: e?.message ?? 'nearby-incidents-error' });
  }
});

// Nearby Businesses
router.get('/businesses/nearby', async (req, res) => {
  try {

    const { rows } = await pool.query(

// Duplicate incident check
router.post('/incidents/duplicates', express.json(), async (req, res) => {
  try {
    const { type, latitude, longitude } = req.body;
    const radius = 150; // 150m duplicate window

    const { rows } = await pool.query(
      `SELECT id,
              earth_distance(ll_to_earth($1,$2), ll_to_earth(lat, lon)) AS meters
       FROM incidents
       WHERE type = $3

    res.status(500).json({ error: e?.message ?? 'duplicate-check-error' });
  }
});

export default router;
```
