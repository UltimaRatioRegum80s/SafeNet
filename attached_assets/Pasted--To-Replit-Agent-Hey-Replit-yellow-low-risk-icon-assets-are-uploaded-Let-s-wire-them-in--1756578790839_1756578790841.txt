🔧 Step 3 — Map markers (pin icons)

client/src/components/MapView.tsx

import L from "leaflet";
import { INCIDENT_TYPES_BY_ID } from "@/shared/incidentTypes";

function markerIconFor(typeId: string) {
  const t = INCIDENT_TYPES_BY_ID[typeId];
  const url = t?.mapIcon ?? `/attached_assets/${typeId}.png`;
  return L.icon({
    iconUrl: url,
    iconSize: [36, 36],
    iconAnchor: [18, 34],
    popupAnchor: [0, -28],
  });
}
