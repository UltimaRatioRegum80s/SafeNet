1) Share a tiny normalizer
// src/lib/incidentsShape.ts
export type Incident = {
  id: string; title: string; createdAt: string; lat: number; lng: number;
  // …other fields you display
};

export function itemsFromFeedPayload(payload: any): Incident[] {
  if (Array.isArray(payload)) return payload as Incident[];
  if (payload && Array.isArray(payload.items)) return payload.items as Incident[];
  return [];
}

2) Map panel: read Feed data from React Query cache
// src/features/map/RecentIncidentsPanel.tsx
import { useEffect, useState } from "react";
import { useQueryClient } from "@tanstack/react-query";
import { itemsFromFeedPayload, type Incident } from "@/lib/incidentsShape";

const FEED_INCIDENTS_KEY = ["/api/incidents"]; // must match Feed page

export default function RecentIncidentsPanel() {
  const qc = useQueryClient();
  const [incidents, setIncidents] = useState<Incident[]>([]);

  // 1) Pull whatever the Feed has already fetched
  useEffect(() => {
    const data = qc.getQueryData(FEED_INCIDENTS_KEY);
    const items = itemsFromFeedPayload(data);
    setIncidents(items);
  }, [qc]);

  // 2) Subscribe to future cache updates from the Feed
  useEffect(() => {
    const unsub = qc.getQueryCache().subscribe((event) => {
      if (event?.query?.queryKey?.toString() === FEED_INCIDENTS_KEY.toString()) {
        const next = itemsFromFeedPayload(event.query.state.data);
        setIncidents(next);
      }
    });
    return () => { unsub?.(); };
  }, [qc]);

  if (!incidents.length) {
    return (
      <section className="card">
        <div className="card-body">No incidents.</div>
      </section>
    );
  }

  return (
    <section className="card">
      <div className="card-body space-y-2">
        {incidents.slice(0, 20).map(i => (
          <div key={i.id} className="rounded-xl bg-neutral-800 p-3">
            <div className="font-medium">{i.title}</div>
            <div className="text-xs opacity-70">
              {new Date(i.createdAt).toLocaleTimeString()}
            </div>
          </div>
        ))}
      </div>
    </section>
  );
}


This does not change any styles/layout. It simply mirrors whatever the Feed already has.
