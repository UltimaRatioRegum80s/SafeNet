B) Add severity chips + “peek/half/full” heights (no drag dependency)
export default function IncidentSheet({ open, onOpenChange, incidents, onSelect }: Props) {
  const [activeSeverity, setActiveSeverity] = useState<Severity | "all">("all");
  const [height, setHeight] = useState<"peek"|"half"|"full">("peek");

  const filtered = useMemo(
    () => incidents.filter(i => (activeSeverity === "all" ? true : i.severity === activeSeverity)),
    [incidents, activeSeverity]
  );

  return (
    <Drawer.Root open={open} onOpenChange={onOpenChange} modal={false} shouldScaleBackground={false}>
      <Drawer.Trigger className="hidden" />
      <Drawer.Portal>
        <Drawer.Overlay className="fixed inset-0 bg-black/20" />
        <Drawer.Content className="fixed inset-x-0 bottom-0 z-30">
          <div className={`mx-auto max-w-md transition-[height] duration-200
            ${height === "full" ? "h-[80vh]" : height === "half" ? "h-[52vh]" : "h-[96px]"}
          `}>
            <div className="mx-3 rounded-t-3xl bg-slate-900 border border-slate-800 shadow-2xl h-full flex flex-col">
              {/* Handle: cycle peek → half → full */}
              <button
                onClick={() => setHeight(h => (h === "peek" ? "half" : h === "half" ? "full" : "peek"))}
                className="flex justify-center pt-2 w-full"
                aria-label="Toggle incidents"
              >
                <div className="h-1.5 w-12 rounded-full bg-white/30" />
              </button>

              {/* Header */}
              <div className="px-4 py-3 flex items-center justify-between">
                <div className="text-base font-semibold">Recent Incidents</div>
                <div className="text-xs text-white/60">{filtered.length} shown</div>
              </div>

              {/* Severity chips */}
              <div className="px-3 pb-2 flex gap-2 overflow-x-auto">
                {(["all","low","medium","high","emergency"] as const).map(sev => (
                  <button
                    key={sev}
                    onClick={() => setActiveSeverity(sev as any)}
                    className={`px-3 py-1.5 rounded-full text-xs font-medium border ${
                      activeSeverity === sev
                        ? "bg-white text-slate-900 border-white"
                        : "bg-transparent text-white/80 border-white/20"
                    }`}
                  >
                    {sev}
                  </button>
                ))}
              </div>

              {/* List */}
              <div className="px-2 pb-3 overflow-y-auto [@supports(height:100svh)]:max-h-[calc(100svh-220px)]">
                {filtered.map(i => (
                  <button
                    key={i.id}
                    onClick={() => onSelect(i.id)}
                    className="w-full text-left mb-2 p-3 rounded-2xl bg-white/5 border border-white/10 hover:bg-white/10 active:scale-[0.99] transition"
                  >
                    <div className="flex items-start gap-3">
                      <div className={`mt-1 size-3 rounded-full ${severityColor(i.severity)}`} />
                      <div className="flex-1 min-w-0">
                        <div className="flex items-center gap-2">
                          <SeverityIcon sev={i.severity} />
                          <div className="font-medium truncate">{i.title}</div>
                        </div>
                        {i.description && (
                          <div className="text-sm text-white/70 line-clamp-2">{i.description}</div>
                        )}
                        <div className="mt-1 flex items-center gap-2 text-xs text-white/60">
                          <SeverityChip sev={i.severity} />
                          <span>{timeAgo(i.created_at)}</span>
                        </div>
                      </div>
                    </div>
                  </button>
                ))}

                {filtered.length === 0 && (
                  <div className="p-6 text-center text-white/60">No incidents.</div>
                )}
              </div>
            </div>
          </div>
        </Drawer.Content>
      </Drawer.Portal>
    </Drawer.Root>
  );
}
