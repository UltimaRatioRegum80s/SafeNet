const controllerRef = useRef<AbortController|null>(null);
const [uploading, setUploading] = useState(false);

async function prepareAndUploadWithUi(files: File[]) {
  controllerRef.current = new AbortController();
  setUploading(true);
  try {
    return await prepareAndUpload(files, controllerRef.current.signal);
  } finally {
    setUploading(false);
  }
}

function cancelUpload() {
  controllerRef.current?.abort();

function pickBestImage(files, target = 400) {
  const byVar = Object.fromEntries(files.map(f => [f.variant, f]));
