client/src/pages/community/Map.tsx — inside your Map component after the Leaflet map is created:

// add once, after `const map = L.map(...)`
const simpleDots = L.layerGroup().addTo(map);
const simpleDotById = new Map<string, L.CircleMarker>();

const severityColor: Record<string, string> = {
  low: "#facc15",      // yellow
  medium: "#f59e0b",   // amber
  high: "#ef4444",     // red
  critical: "#b91c1c", // darker red
};

// GLOBAL helpers for optimistic dots
(window as any).__nnSimpleDot = function addSimpleDot(
  id: string,
  lat: number,
  lng: number,
  severity: string = "low"
) {
  // if it already exists, skip
  if (simpleDotById.has(id)) return;
  const color = severityColor[severity] || "#f59e0b";
  const m = L.circleMarker([lat, lng], {
    radius: 7,
    color,
    weight: 1.5,
    fillColor: color,
    fillOpacity: 0.9,
  });
  m.addTo(simpleDots);
  simpleDotById.set(id, m);
};

(window as any).__nnRemoveSimpleDot = function removeSimpleDot(id: string) {
  const m = simpleDotById.get(id);
  if (!m) return;
  simpleDots.removeLayer(m);
  simpleDotById.delete(id);
};

// provide a safe stub to kill the runtime error you’re seeing
(window as any).refreshClusterFor = (/* _bbox? */) => {};


This layer does not replace your MarkerCluster; it’s just a visual fail-safe.

3) Report submit: create a dot first, then POST (with sane timeouts)

Where you handle the modal Submit (same place that currently toasts “Timeout expired”), replace the handler with this minimal flow:

import { v4 as uuid } from "uuid";

async function getCoordsWithTimeout(ms = 6000) {
  return new Promise<{ lat: number; lng: number }>((resolve, reject) => {
    if (!navigator.geolocation) return reject(new Error("No geolocation"));
    const t = setTimeout(() => reject(new Error("GPS timeout")), ms);
    navigator.geolocation.getCurrentPosition(
      (pos) => {
        clearTimeout(t);
        resolve({ lat: pos.coords.latitude, lng: pos.coords.longitude });
      },
      (err) => {
        clearTimeout(t);
        reject(new Error(err?.message || "GPS failed"));
      },
      { enableHighAccuracy: true, timeout: ms, maximumAge: 0 }
    );
  });
}

async function submitIncidentQuick({
  title,
  description,
  category,
  severity, // "low" | "medium" | "high" | "critical"
}: {
  title: string;
  description?: string;
  category: string;
  severity: string;
}) {
  // 1) get coords (fallback to last map center if GPS blocks)
  let coords: { lat: number; lng: number };
  try {
    coords = await getCoordsWithTimeout(6000);
  } catch {
    const center = (window as any).__nnMap?.getCenter?.();
    if (!center) throw new Error("Location unavailable");
    coords = { lat: center.lat, lng: center.lng };
  }

  // 2) paint an optimistic dot immediately
  const tempId = `temp_${Date.now()}_${Math.random().toString(36).slice(2,7)}`;
  (window as any).__nnSimpleDot?.(tempId, coords.lat, coords.lng, severity);

  // 3) POST with an AbortController timeout (12s)
  const ac = new AbortController();
  const timeout = setTimeout(() => ac.abort(), 12000);
  const res = await fetch("/api/incidents", {
    method: "POST",
    credentials: "include",
    headers: {
      "Content-Type": "application/json",
      "x-idempotency-key": uuid(),
      "x-nn-quick": "1", // harmless header; server can ignore
    },
    body: JSON.stringify({
      title,
      description: description || "",
      category,
      severity,
      latitude: coords.lat,
      longitude: coords.lng,
      // no photos on this quick path
      isAnonymous: true, // or your default
    }),
    signal: ac.signal,
  }).finally(() => clearTimeout(timeout));

  if (!res.ok) {
    // leave the temporary dot for 30s so user still sees something
    setTimeout(() => (window as any).__nnRemoveSimpleDot?.(tempId), 30000);
    throw new Error(await res.text());
  }

  // success → real-time socket will arrive and remove the temp dot via step (2)
  return await res.json();
}


Then call submitIncidentQuick({ title, description, category, severity }) from your modal’s Submit button. Also change any existing calls to refreshClusterFor(...) to the safe optional call:
