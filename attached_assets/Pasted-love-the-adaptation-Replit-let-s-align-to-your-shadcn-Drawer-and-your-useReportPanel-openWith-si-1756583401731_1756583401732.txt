1) New component: 3 severity FABs + bottom sheets

client/src/features/report/CategorySheets.tsx

import { useMemo, useState } from "react";
import { INCIDENT_TYPES, INCIDENT_TYPES_BY_ID } from "@/shared/incidentTypes";
import { useReportPanel } from "@/features/report/useReportPanel";
import {
  Drawer,
  DrawerContent,
  DrawerHeader,
  DrawerTitle,
} from "@/components/ui/drawer"; // shadcn drawer wrapper you already have
import { AlertTriangle, Circle, Flame } from "lucide-react";
import clsx from "clsx";

type Severity = "low" | "medium" | "high";

const COLORS = {
  low:    "from-yellow-400/90 to-yellow-500/90 text-slate-900",
  medium: "from-amber-500/90 to-amber-600/90 text-slate-900",
  high:   "from-red-600/90 to-red-700/90 text-white",
} as const;

function SeverityFab({
  severity,
  label,
  style,
  onClick,
}: {
  severity: Severity;
  label: string;
  style: React.CSSProperties;
  onClick: () => void;
}) {
  const icon =
    severity === "low" ? <Circle className="h-5 w-5" /> :
    severity === "medium" ? <AlertTriangle className="h-5 w-5" /> :
    <Flame className="h-5 w-5" />;

  return (
    <button
      onClick={onClick}
      aria-label={`${label} incidents`}
      className={clsx(
        "pointer-events-auto rounded-full shadow-lg ring-1 ring-white/20",
        "backdrop-blur-md bg-gradient-to-br px-4 py-3 flex items-center gap-2",
        "active:scale-[0.98] transition nn-glass",
        COLORS[severity],
      )}
      style={style}
    >
      {icon}
      <span className="font-semibold">{label}</span>
    </button>
  );
}

function SeverityGrid({
  severity,
  onPick,
}: {
  severity: Severity;
  onPick: (id: string) => void;
}) {
  const items = useMemo(
    () => INCIDENT_TYPES.filter(t => t.severity === severity),
    [severity]
  );
  return (
    <div className="grid grid-cols-3 gap-3 p-4 pb-6">
      {items.map(t => (
        <button
          key={t.id}
          onClick={() => onPick(t.id)}
          className="glass-card rounded-xl p-3 flex flex-col items-center gap-2 hover:bg-white/5 active:scale-[0.98] transition focus:outline-none focus-visible:ring-2 focus-visible:ring-white/40"
        >
          <img
            src={t.buttonIcon ?? "/attached_assets/default.png"}
            alt={t.label}
            className="h-10 w-10"
            loading="lazy"
            decoding="async"
          />
          <span className="text-xs text-center">{t.label}</span>
        </button>
      ))}
    </div>
  );
}

export default function CategorySheets() {
  const [open, setOpen] = useState<Severity | null>(null);
  const rp = useReportPanel();

  const onPick = (id: string) => {
    const t = INCIDENT_TYPES_BY_ID[id];
    if (!t) return;
    setOpen(null);
    // ✅ your hook expects the full incident object
    rp.openWith(t);
  };

  // FAB stack (bottom-right), aware of safe-area; spaced to avoid map controls on right.
  const baseRight = 16; // px; sm screens will be tight
  const safeBottom = "calc(env(safe-area-inset-bottom) + 72px)";

  return (
    <>
      {/* FABs: show on mobile/tablet only, hide on lg+ where desktop grid is used */}
      <div className="fixed inset-0 pointer-events-none lg:hidden z-[55]">
        <SeverityFab
          severity="high"
          label="High"
          onClick={() => setOpen("high")}
          style={{ right: baseRight, bottom: `calc(${safeBottom} + 0px)`, position: "absolute" }}
        />
        <SeverityFab
          severity="medium"
          label="Medium"
          onClick={() => setOpen("medium")}
          style={{ right: baseRight, bottom: `calc(${safeBottom} + 56px)`, position: "absolute" }}
        />
        <SeverityFab
          severity="low"
          label="Low"
          onClick={() => setOpen("low")}
          style={{ right: baseRight, bottom: `calc(${safeBottom} + 112px)`, position: "absolute" }}
        />
      </div>

      {/* Low */}
      <Drawer open={open === "low"} onOpenChange={(v) => setOpen(v ? "low" : null)}>
        <DrawerContent className="glass-surface rounded-t-2xl z-[61]">
          <DrawerHeader>
            <DrawerTitle>Low-risk incidents</DrawerTitle>
          </DrawerHeader>
          <SeverityGrid severity="low" onPick={onPick} />
        </DrawerContent>
      </Drawer>

      {/* Medium */}
      <Drawer open={open === "medium"} onOpenChange={(v) => setOpen(v ? "medium" : null)}>
        <DrawerContent className="glass-surface rounded-t-2xl z-[61]">
          <DrawerHeader>
            <DrawerTitle>Medium-risk incidents</DrawerTitle>
          </DrawerHeader>
          <SeverityGrid severity="medium" onPick={onPick} />
        </DrawerContent>
      </Drawer>

      {/* High */}
      <Drawer open={open === "high"} onOpenChange={(v) => setOpen(v ? "high" : null)}>
        <DrawerContent className="glass-surface rounded-t-2xl z-[61]">
          <DrawerHeader>
            <DrawerTitle>High-risk incidents</DrawerTitle>
          </DrawerHeader>
          <SeverityGrid severity="high" onPick={onPick} />
        </DrawerContent>
      </Drawer>
    </>
  );
}
