// shared/schema.ts
import { z } from "zod";

// Extend your existing insertIncidentSchema
export const insertIncidentSchema = z.object({
  title: z.string().min(3).max(120),
  description: z.string().min(3).max(5000),
  // 👇 changed: coerce strings to numbers, then validate
  lat: z.coerce.number().refine(v => Number.isFinite(v) && Math.abs(v) <= 90, "Invalid latitude"),
  lng: z.coerce.number().refine(v => Number.isFinite(v) && Math.abs(v) <= 180, "Invalid longitude"),
  type: z.string().min(1),
  photoUrl: z.string().url().optional(),
  // ...keep any other fields you already have
});

// Export a tiny normalizer so routes can safely finalize values
export function normalizeIncidentCoords<T extends { lat: number; lng: number }>(parsed: T) {
  let { lat, lng } = parsed;

  // Heuristic swap if they look reversed (mobile autofill glitch)
  if (Math.abs(lat) > 90 && Math.abs(lng) <= 90) {
    [lat, lng] = [lng, lat];
  }
  // Round for DB precision & stable clustering
  lat = Number(lat.toFixed(6));
  lng = Number(lng.toFixed(6));

  return { ...parsed, lat, lng };
}


This preserves your existing import sites and just improves robustness.

import { io } from "socket.io-client";
import { QueryClient } from "@tanstack/react-query";

export const socket = io("/", {
  transports: ["websocket", "polling"],
  reconnection: true,
  reconnectionAttempts: Infinity,
  reconnectionDelay: 500,
  reconnectionDelayMax: 5000,
  timeout: 20000,
  autoConnect: true,
});

// Call once at app bootstrap (e.g., in App.tsx) after you create queryClient
export function wireIncidentSocketToQueryCache(queryClient: QueryClient) {
  const upsert = (inc: any) => {
    queryClient.setQueryData<any[]>(["incidents"], (old = []) => {
      // de-dupe by id; put newest first
      const withoutDup = old.filter(i => i.id !== inc.id && i.id !== inc?.tempId && !i.__optimistic);
      return [inc, ...withoutDup];
    });
  };

  socket.on("incident:new", upsert);
}

4) client/src/hooks/useCreateIncident.ts — optimistic create using existing React Query
// client/src/hooks/useCreateIncident.ts
import { useMutation, useQueryClient } from "@tanstack/react-query";

export function useCreateIncident() {
  const qc = useQueryClient();
  return useMutation({
    mutationFn: async (payload: any) => {
      const r = await fetch("/api/incidents", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload),
      });
      if (!r.ok) throw new Error("Failed to create incident");
      return r.json();
    },
    onMutate: async (vars: any) => {
      await qc.cancelQueries({ queryKey: ["incidents"] });
      const previous = qc.getQueryData<any[]>(["incidents"]) || [];
      const optimistic = {
        ...vars,
        id: `temp-${Date.now()}`,
        __optimistic: true,
      };
      qc.setQueryData(["incidents"], [optimistic, ...previous]);
      return { previous };
    },
    onError: (_err, _vars, ctx) => {
      if (ctx?.previous) qc.setQueryData(["incidents"], ctx.previous);
    },
    onSuccess: (created) => {
      qc.setQueryData<any[]>(["incidents"], (old = []) => {
        const filtered = old.filter(i => i.id !== created.id && !i.__optimistic);
        return [created, ...filtered];
      });
    },
    onSettled: () => {
      qc.invalidateQueries({ queryKey: ["incidents"] });
    },
  });
}

6) client/src/components/InteractiveMap.tsx — stable refs, resize fixes, icon helper
// client/src/components/InteractiveMap.tsx
import { useEffect, useMemo, useRef } from "react";
import { useQuery } from "@tanstack/react-query";
import L, { Map as LeafletMap } from "leaflet";
import "leaflet/dist/leaflet.css";
import "leaflet.markercluster";
import "leaflet.markercluster/dist/MarkerCluster.css";
import "leaflet.markercluster/dist/MarkerCluster.Default.css";
import { getIncidentIcon } from "@/lib/incidentIcons";

type Incident = {
  id: string;
  lat: number;
  lng: number;
  type: string;
  title?: string;
};

export default function InteractiveMap() {
  const mapRef = useRef<LeafletMap | null>(null);
  const clusterRef = useRef<L.MarkerClusterGroup | null>(null);

  const { data: incidents = [] } = useQuery<Incident[]>({
    queryKey: ["incidents"],
    queryFn: async () => (await fetch("/api/incidents")).json(),
    staleTime: 30_000,
    refetchOnWindowFocus: true,
    refetchOnReconnect: true,
    networkMode: "always",
  });

  // Init map + cluster once
  useEffect(() => {
    if (mapRef.current) return;

    const map = L.map("nn-map", {
      center: [-33.9249, 18.4241],
      zoom: 12,
      zoomControl: false,
    });

    L.tileLayer("https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png", {
      maxZoom: 19,
      attribution: "&copy; OpenStreetMap",
    }).addTo(map);

    const cluster = L.markerClusterGroup({
      chunkedLoading: true,
      removeOutsideVisibleBounds: false,
      spiderfyOnMaxZoom: true,
    });
    cluster.addTo(map);

    // Mobile layout & orientation resiliency
    const invalidate = () => setTimeout(() => map.invalidateSize(), 0);
    window.addEventListener("resize", invalidate);
    window.addEventListener("orientationchange", invalidate);
    const container = document.getElementById("nn-map");
    const ro = new ResizeObserver(invalidate);
    if (container) ro.observe(container);

    mapRef.current = map;
    clusterRef.current = cluster;

    return () => {
      window.removeEventListener("resize", invalidate);
      window.removeEventListener("orientationchange", invalidate);
      ro.disconnect();
      map.remove();
      mapRef.current = null;
      clusterRef.current = null;
    };
  }, []);

  // Render markers whenever cache changes (includes socket upserts)
  useEffect(() => {
    if (!mapRef.current || !clusterRef.current) return;
    const map = mapRef.current;
    const cluster = clusterRef.current;

    cluster.clearLayers();

    for (const inc of incidents) {
      if (!Number.isFinite(inc.lat) || !Number.isFinite(inc.lng)) continue;

      const marker = L.marker([inc.lat, inc.lng], {
        icon: getIncidentIcon(inc.type),
        keyboard: false,
      }).bindPopup(`<strong>${inc.title ?? "Incident"}</strong>`);

      cluster.addLayer(marker);
    }

    // Fit bounds on first load or when markers appear
    if (incidents.length > 0) {
      const group = L.featureGroup(
        incidents.map((i) => L.marker([i.lat, i.lng]))
      );
      try {
        map.fitBounds(group.getBounds(), { maxZoom: 15, padding: [30, 30] });
      } catch {}
    } else {
      map.setView([-33.9249, 18.4241], 12);
    }

    setTimeout(() => map.invalidateSize(), 0);
  }, [incidents]);

  return <div id="nn-map" className="h-[calc(100vh-8rem)] w-full rounded-2xl shadow-md" />;
}
