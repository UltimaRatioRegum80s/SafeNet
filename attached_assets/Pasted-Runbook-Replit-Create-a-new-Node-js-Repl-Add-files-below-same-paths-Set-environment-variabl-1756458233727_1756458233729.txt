server/index.ts (Express + JWT, httpOnly refresh cookie)
import "dotenv/config";
import express from "express";
import cors from "cors";
import cookieParser from "cookie-parser";
import jwt from "jsonwebtoken";
import { nanoid } from "nanoid";
import path from "node:path";

const app = express();
app.use(express.json());
app.use(cookieParser());
app.use(cors({
  origin: true,
  credentials: true
}));

// ---- In-memory refresh session store (swap for Drizzle/Postgres in NN)
type Session = { rid: string; userId: string; expiresAt: number; revoked?: boolean };
const sessions = new Map<string, Session>();

const ACCESS_TTL_SEC = 60;         // short-lived access token
const REFRESH_TTL_SEC = 60 * 60 * 24 * 7; // 7 days

const ACCESS_SECRET = process.env.JWT_ACCESS_SECRET!;
const REFRESH_SECRET = process.env.JWT_REFRESH_SECRET!;

function signAccessToken(userId: string) {
  return jwt.sign({ sub: userId }, ACCESS_SECRET, { expiresIn: ACCESS_TTL_SEC });
}
function signRefreshToken(userId: string, rid: string) {
  return jwt.sign({ sub: userId, rid }, REFRESH_SECRET, { expiresIn: REFRESH_TTL_SEC });
}

function setRefreshCookie(res: express.Response, token: string) {
  const isProd = process.env.NODE_ENV === "production";
  res.cookie("nn_refresh", token, {
    httpOnly: true,
    secure: isProd,           // set true on HTTPS (Replit webview is HTTPS)
    sameSite: "lax",
    path: "/api/auth"
  });
}

function clearRefreshCookie(res: express.Response) {
  res.clearCookie("nn_refresh", { path: "/api/auth" });
}

// ---- Demo user
const DEMO_USER = { id: "u_demo", email: "demo@nabor.net", name: "Demo Neighbor" };

// ---- Auth routes
app.post("/api/auth/login", async (req, res) => {
  const { email, password } = req.body ?? {};
  if (email !== "demo@nabor.net" || password !== "demo123") {
    return res.status(401).json({ error: "Invalid credentials" });
  }
  const rid = nanoid();
  const refresh = signRefreshToken(DEMO_USER.id, rid);
  const access = signAccessToken(DEMO_USER.id);

  sessions.set(rid, { rid, userId: DEMO_USER.id, expiresAt: Date.now() + REFRESH_TTL_SEC * 1000 });

  setRefreshCookie(res, refresh);
  return res.json({ accessToken: access, user: { id: DEMO_USER.id, email: DEMO_USER.email, name: DEMO_USER.name } });
});

app.post("/api/auth/refresh", async (req, res) => {
  const token = req.cookies["nn_refresh"];
  if (!token) return res.status(401).json({ error: "No refresh cookie" });
  try {
    const payload = jwt.verify(token, REFRESH_SECRET) as { sub: string; rid: string; exp: number };
    const s = sessions.get(payload.rid);
    if (!s || s.revoked) return res.status(401).json({ error: "Session invalid" });

    // rotate
    s.revoked = true;
    const newRid = nanoid();
    const newRefresh = signRefreshToken(payload.sub, newRid);
    sessions.set(newRid, { rid: newRid, userId: payload.sub, expiresAt: Date.now() + REFRESH_TTL_SEC * 1000 });

    setRefreshCookie(res, newRefresh);
    const newAccess = signAccessToken(payload.sub);
    return res.json({ accessToken: newAccess, user: { id: DEMO_USER.id, email: DEMO_USER.email, name: DEMO_USER.name } });
  } catch {
    clearRefreshCookie(res);
    return res.status(401).json({ error: "Invalid refresh" });
  }
});

app.post("/api/auth/logout", (req, res) => {
  const token = req.cookies["nn_refresh"];
  if (token) {
    try {
      const payload = jwt.verify(token, REFRESH_SECRET) as any;
      const s = sessions.get(payload.rid);
      if (s) s.revoked = true;
    } catch {}
  }
  clearRefreshCookie(res);
  return res.json({ ok: true });
});

// Example protected resource
function authMiddleware(req: any, res: any, next: any) {
  const header = req.headers.authorization || "";
  const token = header.startsWith("Bearer ") ? header.slice(7) : null;
  if (!token) return res.status(401).json({ error: "No token" });
  try {
    const payload = jwt.verify(token, ACCESS_SECRET) as any;
    req.userId = payload.sub;
    next();
  } catch {
    return res.status(401).json({ error: "Invalid token" });
  }
}

app.get("/api/me", authMiddleware, (req, res) => {
  return res.json({ user: DEMO_USER });
});

const PORT = 3000;
app.listen(PORT, () => {
  console.log(`API on http://localhost:${PORT}`);
});

src/auth/store.ts (Zustand, persisted but no tokens saved)
import { create } from "zustand";

type User = { id: string; email: string; name: string } | null;

type AuthState = {
  user: User;
  accessToken: string | null; // kept in memory only
  setAuth: (u: User, t: string) => void;
  clear: () => void;
  hydrated: boolean;
  setHydrated: () => void;
};

export const useAuth = create<AuthState>((set) => ({
  user: null,
  accessToken: null,
  setAuth: (user, token) => set({ user, accessToken: token }),
  clear: () => set({ user: null, accessToken: null }),
  hydrated: false,
  setHydrated: () => set({ hydrated: true })
}));

// Cross-tab sync
const ch = new BroadcastChannel("nn-auth");
export function broadcastAuth(event: "login" | "logout", payload?: any) {
  ch.postMessage({ event, payload });
}
ch.onmessage = (e) => {
  const { event, payload } = e.data || {};
  if (event === "login") {
    useAuth.getState().setAuth(payload.user, payload.accessToken);
  }
  if (event === "logout") {
    useAuth.getState().clear();
  }
};

src/api/axios.ts (axios client with 401 refresh & silent boot refresh)
import axios from "axios";
import { useAuth, broadcastAuth } from "../auth/store";

export const api = axios.create({
  baseURL: "/api",
  withCredentials: true // needed for httpOnly refresh cookie
});

// attach access token
api.interceptors.request.use((config) => {
  const token = useAuth.getState().accessToken;
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

let refreshing: Promise<void> | null = null;

async function doRefresh() {
  const res = await api.post("/auth/refresh");
  const { accessToken, user } = res.data;
  useAuth.getState().setAuth(user, accessToken);
}

// on 401, try refresh once
api.interceptors.response.use(
  (r) => r,
  async (error) => {
    const original = error.config;
    if (error?.response?.status === 401 && !original._retry) {
      original._retry = true;
      if (!refreshing) refreshing = doRefresh().finally(() => (refreshing = null));
      await refreshing;
      return api(original);
    }
    return Promise.reject(error);
  }
);

// Silent refresh on app boot
export async function silentBootRefresh() {
  try {
    await doRefresh();
    broadcastAuth("login", { user: useAuth.getState().user, accessToken: useAuth.getState().accessToken });
  } catch {
    // no active session, ignore
  } finally {
    useAuth.getState().setHydrated();
  }
}

src/App.tsx (UI + flows)
import { useEffect, useState } from "react";
import { useAuth, broadcastAuth } from "./auth/store";
import { api, silentBootRefresh } from "./api/axios";

export default function App() {
  const { user, accessToken, setAuth, clear, hydrated } = useAuth();
  const [email, setEmail] = useState("demo@nabor.net");
  const [password, setPassword] = useState("demo123");
  const [me, setMe] = useState<any>(null);
  const [status, setStatus] = useState<string>("");

  useEffect(() => {
    // do not logout on beforeunload — requirement: session persists naturally
    silentBootRefresh();
  }, []);

  async function login() {
    setStatus("Logging in…");
    const res = await api.post("/auth/login", { email, password });
    setAuth(res.data.user, res.data.accessToken);
    broadcastAuth("login", { user: res.data.user, accessToken: res.data.accessToken });
    setStatus("Logged in");
  }

  async function logout() {
    await api.post("/auth/logout");
    clear();
    broadcastAuth("logout");
    setMe(null);
    setStatus("Logged out");
  }

  async function fetchMe() {
    try {
      const res = await api.get("/me");
      setMe(res.data.user);
      setStatus("Fetched /me");
    } catch (e: any) {
      setStatus(`Error: ${e?.response?.status || "?"}`);
    }
  }

  return (
    <div style={{ fontFamily: "Inter, system-ui", padding: 24, maxWidth: 720, margin: "0 auto" }}>
      <h1>NaborNet Auth Persistence Repro</h1>
      <p>HTTP-only refresh cookie · short-lived access token · silent refresh · cross-tab sync</p>

      <section style={{ marginTop: 16, padding: 16, border: "1px solid #eee", borderRadius: 12 }}>
        <h2>Auth</h2>
        {user ? (
          <>
            <p>Logged in as <b>{user.email}</b></p>
            <button onClick={logout}>Logout</button>
          </>
        ) : (
          <>
            <input value={email} onChange={(e) => setEmail(e.target.value)} placeholder="email" />
            <input type="password" value={password} onChange={(e) => setPassword(e.target.value)} placeholder="password" />
            <button onClick={login}>Login</button>
          </>
        )}
      </section>

      <section style={{ marginTop: 16, padding: 16, border: "1px solid #eee", borderRadius: 12 }}>
        <h2>Protected Request</h2>
        <button onClick={fetchMe} disabled={!accessToken && !hydrated}>GET /api/me</button>
        <pre>{me ? JSON.stringify(me, null, 2) : "No data yet"}</pre>
      </section>

      <section style={{ marginTop: 16 }}>
        <small>Status: {status} | Hydrated: {String(hydrated)} | Has access: {String(!!accessToken)}</small>
      </section>
    </div>
  );
}
