export function AppChrome({ title, rightActions, hideSubnav, children }: AppChromeProps) {
  const topRef = React.useRef<HTMLDivElement>(null);
  const subRef = React.useRef<HTMLDivElement>(null);
  const topH = useMeasured(topRef);
  const subH = useMeasured(subRef);


  return (
    <div className="min-h-screen bg-slate-950 text-slate-100">
      {/* Fixed Top */}
      <div ref={topRef} className="fixed inset-x-0 top-0 z-50" style={{ paddingTop: "env(safe-area-inset-top)" }}>
        <TopBar title={title} right={rightActions} />
        {!hideSubnav && <div ref={subRef as any}><PrimaryNav /></div>}
      </div>


      {/* Content with dynamic padding */}
      <main
        className="relative"
        style={{ paddingTop: topH + (hideSubnav ? 0 : subH), minHeight: "100dvh" }}
      >
        {children}
      </main>
    </div>
  );
}


// ================== MapChrome.tsx (map-special layout) ==================
// Full-bleed map: keeps only TopBar, hides subnav for more real estate.


export function MapChrome({ title = "Map", children }: { title?: string; children: React.ReactNode }) {
  return (
    <AppChrome title={title} hideSubnav>
      <div className="fixed inset-0 top-[--top-offset]">{children}</div>
    </AppChrome>
  );
}

totally aligned, @RDM — we can (and should) have a uniform menu across all pages with a single exception for Maps.

I just added to the canvas:

AppChrome — a global layout that renders the exact header/subnav shown in your screenshots (title + icons row, then “Home / Report / Map / Feed”). It fixes them to the top, measures heights, and pads content so nothing is hidden.

PrimaryNav — the uniform subnav component with active-state highlighting.

TopBar — the header with Share/Search/Bell/Theme icons.

MapChrome — a map-specific variant that hides the subnav (your one exception) to give max screen space.

There’s also a React Router v6 usage example so every non-map route inherits the same chrome, while /map uses the special layout.

what this gives you

Same look/behavior for Home, Report, Feed, Chat, etc.

Maps keeps only the header (no subnav row), matching your requirement.

Safe-area compatible, responsive, and minimal disruption to your current routing.
