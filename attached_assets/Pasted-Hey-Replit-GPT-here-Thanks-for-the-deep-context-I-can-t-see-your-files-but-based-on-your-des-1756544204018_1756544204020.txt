1) useIncidentStore (keep useAuthStore separate)

Don’t consolidate with auth. Keep a small, focused store for UI state (no network data).

// client/src/state/useIncidentStore.ts
import { create } from "zustand";

export type Severity = "low" | "medium" | "high" | "emergency";
export type TimeRange = "all" | "24h" | "7d" | "30d";
export type BBox = { north:number; south:number; east:number; west:number } | null;

type S = {
  selectedId: string | null;
  setSelectedId: (id: string | null) => void;

  severity: Severity | "all";
  setSeverity: (s: S["severity"]) => void;

  timeRange: TimeRange;
  setTimeRange: (t: TimeRange) => void;

  inViewOnly: boolean;
  toggleInViewOnly: () => void;

  bbox: BBox;
  setBBox: (b: BBox) => void;
};

export const useIncidentStore = create<S>((set) => ({
  selectedId: null,
  setSelectedId: (id) => set({ selectedId: id }),

  severity: "all",
  setSeverity: (s) => set({ severity: s }),

  timeRange: "all",
  setTimeRange: (t) => set({ timeRange: t }),

  inViewOnly: true,
  toggleInViewOnly: () => set((s) => ({ inViewOnly: !s.inViewOnly })),

  bbox: null,
  setBBox: (b) => set({ bbox: b }),
}));


Keep React Query + Socket.IO as your data layers. Store only holds UI state (selection, filters, bbox). It avoids duplication & caching bugs.

2) Wire bbox into the store (Map page)

Debounce updates; only write after move/zoom settles.

// client/src/pages/community/Map.tsx (excerpt)
import { useIncidentStore } from "@/state/useIncidentStore";

const { setBBox } = useIncidentStore.getState(); // or via hook in component

function wireBBox(map: L.Map) {
  let t: any;
  const push = () => {
    const b = map.getBounds();
    setBBox({ north: b.getNorth(), south: b.getSouth(), east: b.getEast(), west: b.getWest() });
  };
  const debounced = () => { clearTimeout(t); t = setTimeout(push, 180); };

  map.whenReady(push);
  map.on("moveend zoomend load", debounced);
}


Call wireBBox(mapRef.current!) right after map init.

4) Derived list for Sidebar/Sheet (no server change)

Compute once per view using store state + your queried/socket data.

// in a component that has `incidents` already loaded
import { useIncidentStore } from "@/state/useIncidentStore";

const { bbox, severity, timeRange, inViewOnly } = useIncidentStore();
const items = useMemo(() => {
  const within = (lat:number,lng:number) =>
    !inViewOnly || !bbox ? true :
    lat <= bbox.north && lat >= bbox.south && lng <= bbox.east && lng >= bbox.west;

  const withinTime = (iso?:string) => {
    if (timeRange === "all" || !iso) return true;
    const ms = Date.now() - new Date(iso).getTime();
    const lim = timeRange === "24h" ? 864e5 : timeRange === "7d" ? 7*864e5 : 30*864e5;
    return ms <= lim;
  };

  return incidents.filter(i =>
    (severity === "all" || i.severity === severity) &&
    within(i.position.lat, i.position.lng) &&
    withinTime(i.created_at)
  );
}, [incidents, bbox, severity, timeRange, inViewOnly]);
