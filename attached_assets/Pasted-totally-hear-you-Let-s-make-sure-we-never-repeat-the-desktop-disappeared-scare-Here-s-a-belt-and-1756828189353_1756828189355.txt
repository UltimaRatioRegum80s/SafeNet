export function readFlag(name: string, def = '0') {
  const url = new URL(location.href);
  const q = url.searchParams.get(name);
  if (q === '1' || q === '0') return q;
  const ls = localStorage.getItem(name);
  if (ls === '1' || ls === '0') return ls;
  return def;
}

export const Flags = {
  fixMobileMarkers: readFlag('NN_FIX_MOBILE_MARKERS', '0') === '1',
};

Gated fallback (replace only the circleMarker branch)

// pseudocode located in incidentIcons.ts (or your marker factory)
import { Flags } from '@/lib/flags';
import { isLikelyMobile } from '@/lib/platform';

let diag = { created: 0, added: 0, clusterChildrenAfter: 0 };

function buildFallbackDivIcon({ size = 14, color = '#2563eb' } = {}) {
  const r = size / 2;
  return L.divIcon({
    className: 'nn-divicon-wrapper',
    html: `<div style="width:${size}px;height:${size}px;border-radius:9999px;background:rgba(37,99,235,0.15);border:2px solid ${color};box-shadow:0 1px 2px rgba(0,0,0,.25)"></div>`,
    iconSize: [size, size],
    iconAnchor: [r, r],
    popupAnchor: [0, -r],
  });
}

// inside addIncidentToCluster(inc)
const icon = typeof getIncidentIcon === 'function' ? getIncidentIcon(inc.type) : null;

let layer: L.Layer;

if (icon) {
  // ✅ existing desktop path (unchanged)
  layer = L.marker(ll, { icon, title: inc.title ?? inc.type });
} else if (Flags.fixMobileMarkers && isLikelyMobile) {
  // 🛡️ gated mobile fallback → still a Marker (cluster-compatible)
  layer = L.marker(ll, {
    icon: buildFallbackDivIcon({ size: 14, color: '#2563eb' }),
    title: inc.title ?? inc.type,
  });
} else {
  // 🚫 legacy fallback retained but not used when flag is on
  layer = L.circleMarker(ll, { radius: 7, color: '#2563eb' });
}

// diagnostics (dev only)
try {
  diag.created++;
  clusterRef.current.addLayer(layer);
  diag.added++;
  // sample the child count after first batch
  if (diag.created === 10) {
    const children = clusterRef.current.getLayers().length;
    diag.clusterChildrenAfter = children;
    if (Flags.fixMobileMarkers && isLikelyMobile && children === 0) {
      console.warn('[NN] Canary: no children after 10 markers. Disabling fallback for this session.');
      localStorage.setItem('NN_FIX_MOBILE_MARKERS', '0'); // kill switch
    }
  }
} catch (e) {
  console.error('[NN] addLayer error', e);
}

import { Flags } from '@/lib/flags';
import { isLikelyMobile } from '@/lib/platform';

// small helper (local to this file)
function buildFallbackDivIcon({ size = 14, color = '#2563eb' } = {}) {
  const r = size / 2;
  return L.divIcon({
    className: 'nn-divicon-wrapper',
    html: `<div style="width:${size}px;height:${size}px;border-radius:9999px;background:rgba(37,99,235,0.15);border:2px solid ${color};box-shadow:0 1px 2px rgba(0,0,0,.25)"></div>`,
    iconSize: [size, size],
    iconAnchor: [r, r],
    popupAnchor: [0, -r],
  });
}

// wherever you currently decide the layer type:
function addIncidentToCluster(inc: any) {
  if (!clusterRef.current) return;

  const ll = L.latLng(inc.lat, inc.lng);
  const icon = (typeof getIncidentIcon === 'function') ? getIncidentIcon(inc.type) : null;

  let layer: L.Layer;

  if (icon) {
    // ✅ Desktop/icon path — unchanged
    layer = L.marker(ll, { icon, title: inc.title ?? inc.type });
  } else if (Flags.fixMobileMarkers && isLikelyMobile) {
    // 🛡️ Gated mobile fallback — still an L.Marker (cluster-compatible)
    layer = L.marker(ll, {
      icon: buildFallbackDivIcon({ size: 14, color: '#2563eb' }),
      title: inc.title ?? inc.type,
    });
  } else {
    // 🚫 Legacy fallback retained (not used when flag is ON); keep for parity
    layer = L.circleMarker(ll, { radius: 7, color: '#2563eb' });
  }

  // existing popup/bind logic stays the same
  if (inc.popupHtml) layer.bindPopup(inc.popupHtml, { maxWidth: 320 });

  // add to cluster
  clusterRef.current.addLayer(layer);
}
