2) Shared IncidentList (reuse in Sidebar + Sheet)
// client/src/components/IncidentList.tsx
import { timeAgo } from "@/lib/utils";
import { SeverityIcon, SeverityChip } from "@/components/SeverityUI";
import { severityColor } from "@/lib/severity";
import { useIncidentStore } from "@/state/useIncidentStore";

type IncidentRow = {
  id: string;
  title: string;
  description?: string;
  severity: "low"|"medium"|"high"|"emergency";
  created_at?: string;
  position: { lat:number; lng:number };
};

export default function IncidentList({
  items,
  onSelect,
  dense = false,
}: {
  items: IncidentRow[];
  onSelect: (id: string) => void;
  dense?: boolean;
}) {
  const selectedId = useIncidentStore(s => s.selectedId);

  return (
    <div className={`px-2 pb-3 overflow-y-auto ${dense ? "space-y-1" : ""}`}>
      {items.map(i => {
        const isActive = i.id === selectedId;
        return (
          <button
            key={i.id}
            onClick={() => onSelect(i.id)}
            aria-selected={isActive}
            className={`w-full text-left ${dense ? "p-2" : "p-3"} mb-2 rounded-2xl border transition
              ${isActive ? "bg-white/15 border-white/30" : "bg-white/5 border-white/10 hover:bg-white/10"}
            `}
          >
            <div className="flex items-start gap-3">
              <div className={`mt-1 size-3 rounded-full ${severityColor(i.severity)}`} />
              <div className="flex-1 min-w-0">
                <div className="flex items-center gap-2">
                  <SeverityIcon sev={i.severity} />
                  <div className="font-medium truncate">{i.title}</div>
                </div>
                {i.description && (
                  <div className="text-sm text-white/70 line-clamp-2">{i.description}</div>
                )}
                <div className="mt-1 flex items-center gap-2 text-xs text-white/60">
                  <SeverityChip sev={i.severity} />
                  <span>{timeAgo(i.created_at)}</span>
                </div>
              </div>
            </div>
          </button>
        );
      })}
      {items.length === 0 && (
        <div className="p-6 text-center text-white/60">No incidents.</div>
      )}
    </div>
  );
}

BBox “{}” error on init
Guard updates until the map is ready:

// Map.tsx (bbox wiring)
function wireBBox(map: L.Map) {
  let t:any;
  const push = () => {
    try {
      if (!(map as any)._loaded) return;
      const b = map.getBounds();
      setBBox({ north:b.getNorth(), south:b.getSouth(), east:b.getEast(), west:b.getWest() });
    } catch { /* swallow transient during init */ }
  };
  const debounced = () => { clearTimeout(t); t = setTimeout(push, 180); };

  map.whenReady(push);
  map.on("load", debounced);
  map.on("moveend", debounced);
  map.on("zoomend", debounced);
}
