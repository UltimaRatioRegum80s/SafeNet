A) Robust device location hook

// client/src/lib/location/useDeviceLocation.ts
import { useEffect, useRef } from "react";
import { useLocationStore } from "@/stores/locationStore";

type LocationStatus = "ok" | "stale" | "denied" | "unavailable";
const MAX_AGE_MS = 10 * 60 * 1000;

export function useDeviceLocation() {
  const setLocation = useLocationStore(s => s.setCurrentLocation);
  const setLastGood = useLocationStore(s => s.setLastGoodLocation);
  const setStatus = useLocationStore(s => s.setLocationStatus);
  const watchRef = useRef<number | null>(null);

  useEffect(() => {
    if (!("geolocation" in navigator)) {
      setStatus("unavailable");
      return;
    }

    const onSuccess = (pos: GeolocationPosition) => {
      const { latitude: lat, longitude: lng } = pos.coords;
      if (Number.isFinite(lat) && Number.isFinite(lng)) {
        const now = Date.now();
        setLocation({ lat, lng, timestamp: now });
        setLastGood({ lat, lng, timestamp: now });
        setStatus("ok");
      } else {
        setStatus("unavailable");
      }
    };

    const onError = (err: GeolocationPositionError) => {
      setStatus(err.code === err.PERMISSION_DENIED ? "denied" : "unavailable");
    };

    watchRef.current = navigator.geolocation.watchPosition(onSuccess, onError, {
      enableHighAccuracy: true,
      timeout: 15000,
      maximumAge: 5000,
    });

    return () => {
      if (watchRef.current != null) {
        navigator.geolocation.clearWatch(watchRef.current);
      }
    };
  }, []);
}

export function pickQueryLocation(store: {
  currentLocation?: { lat: number; lng: number; timestamp: number } | null;
  lastGoodLocation?: { lat: number; lng: number; timestamp: number } | null;
}) {
  const now = Date.now();
  if (store.currentLocation && now - store.currentLocation.timestamp <= MAX_AGE_MS)
    return store.currentLocation;
  if (store.lastGoodLocation && now - store.lastGoodLocation.timestamp <= MAX_AGE_MS)
    return store.lastGoodLocation;
  return null;
}
