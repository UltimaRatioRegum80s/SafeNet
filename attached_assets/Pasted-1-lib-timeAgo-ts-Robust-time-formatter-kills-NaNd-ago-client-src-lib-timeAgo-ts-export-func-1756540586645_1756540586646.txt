1) lib/timeAgo.ts

Robust time formatter (kills NaNd ago).

// client/src/lib/timeAgo.ts
export function timeAgo(iso?: string) {
  if (!iso) return "Just now";
  const d = new Date(iso);
  if (Number.isNaN(d.getTime())) return "Unknown time";
  const s = Math.floor((Date.now() - d.getTime()) / 1000);
  if (s < 60) return `${s}s ago`;
  const m = Math.floor(s / 60);
  if (m < 60) return `${m}m ago`;
  const h = Math.floor(m / 60);
  if (h < 24) return `${h}h ago`;
  const days = Math.floor(h / 24);
  return `${days}d ago`;
}

3) components/IncidentSheet.tsx

Uses vaul for a draggable bottom sheet with peek / half / full and the improved incident cards.

// client/src/components/IncidentSheet.tsx
import * as React from "react";
import { Drawer } from "vaul"; // your project already uses vaul
import { Clock, MapPin, AlertTriangle, Siren, HandHelping } from "lucide-react";
import { Severity, severityColor, severityLabel } from "@/lib/severity";
import { timeAgo } from "@/lib/timeAgo";

export type Incident = {
  id: string;
  title: string;
  description?: string;
  severity: Severity;
  lat: number;
  lng: number;
  createdAt: string; // ISO
};

type Props = {
  open: boolean;
  onOpenChange: (v: boolean) => void;
  incidents: Incident[];
  activeSeverity: Severity | "all";
  setActiveSeverity: (s: Severity | "all") => void;
  onSelect: (id: string) => void; // center map + highlight
};

export function IncidentSheet({
  open,
  onOpenChange,
  incidents,
  activeSeverity,
  setActiveSeverity,
  onSelect,
}: Props) {
  const filtered = incidents.filter(i => activeSeverity === "all" ? true : i.severity === activeSeverity);

  return (
    <Drawer.Root open={open} onOpenChange={onOpenChange} modal={false} shouldScaleBackground={false}>
      {/* We use a hidden trigger – control open from parent */}
      <Drawer.Trigger className="hidden" />
      <Drawer.Portal>
        <Drawer.Overlay className="fixed inset-0 bg-black/20" />
        <Drawer.Content
          className="fixed inset-x-0 bottom-0 z-30"
          style={{ contain: "content" }}
        >
          <div className="mx-auto max-w-md">
            <div className="mx-3 rounded-t-3xl bg-slate-900 border border-slate-800 shadow-2xl">
              {/* Handle */}
              <button
                className="flex justify-center pt-2 w-full"
                onClick={() => onOpenChange(!open)}
                aria-label="Toggle incidents"
              >
                <div className="h-1.5 w-12 rounded-full bg-white/30" />
              </button>

              {/* Header */}
              <div className="px-4 py-3 flex items-center justify-between">
                <div className="text-base font-semibold">Recent Incidents</div>
                <div className="text-xs text-white/60">{filtered.length} shown</div>
              </div>

              {/* Severity chips */}
              <div className="px-3 pb-2 flex gap-2 overflow-x-auto">
                {(["all","low","medium","high","emergency"] as const).map(s => (
                  <button
                    key={s}
                    onClick={() => setActiveSeverity(s)}
                    className={`px-3 py-1.5 rounded-full text-xs font-medium border ${
                      activeSeverity === s
                        ? "bg-white text-slate-900 border-white"
                        : "bg-transparent text-white/80 border-white/20"
                    }`}
                  >
                    {s}
                  </button>
                ))}
              </div>

              {/* List */}
              <div className="max-h-[60vh] overflow-y-auto px-2 pb-3">
                {filtered.map(i => (
                  <button
                    key={i.id}
                    onClick={() => onSelect(i.id)}
                    className="w-full text-left mb-2 p-3 rounded-2xl bg-white/5 border border-white/10 hover:bg-white/10 active:scale-[0.99] transition"
                  >
                    <div className="flex items-start gap-3">
                      <div className={`mt-1 size-3 rounded-full ${severityColor(i.severity)}`} />
                      <div className="flex-1 min-w-0">
                        <div className="flex items-center gap-2">
                          <SeverityIcon sev={i.severity} />
                          <div className="font-medium truncate">{i.title}</div>
                        </div>
                        {i.description && (
                          <div className="text-sm text-white/70 line-clamp-2">{i.description}</div>
                        )}
                        <div className="mt-1 flex items-center gap-2 text-xs text-white/60">
                          <span className="px-2 py-0.5 rounded-full bg-white/10 border border-white/10">
                            {severityLabel(i.severity)}
                          </span>
                          <Clock className="w-3 h-3" />
                          <span>{timeAgo(i.createdAt)}</span>
                        </div>
                      </div>
                    </div>
                  </button>
                ))}
                {filtered.length === 0 && (
                  <div className="p-6 text-center text-white/60">No incidents.</div>
                )}
              </div>
            </div>
          </div>
        </Drawer.Content>
      </Drawer.Portal>
    </Drawer.Root>
  );
}

function SeverityIcon({ sev }: { sev: Severity }) {
  const cls = "w-4 h-4 text-white/80";
  switch (sev) {
    case "emergency": return <Siren className={cls} />;
    case "high": return <AlertTriangle className={cls} />;
    case "medium": return <MapPin className={cls} />;
    default: return <HandHelping className={cls} />;
  }
}

4) components/MapControlsFAB.tsx

Floating controls that drive Leaflet (no plugin required).

// client/src/components/MapControlsFAB.tsx
import * as React from "react";
import { Plus, Minus, Crosshair } from "lucide-react";
import type { Map as LeafletMap } from "leaflet";

export default function MapControlsFAB({
  map,
  getUserCenter,
}: {
  map: LeafletMap | null;
  getUserCenter?: () => [number, number] | null; // optional geolocate hook
}) {
  const zoomIn = () => map?.zoomIn(1);
  const zoomOut = () => map?.zoomOut(1);
  const recenter = () => {
    const c = getUserCenter?.();
    if (map && c) map.setView({ lat: c[1], lng: c[0] }, Math.max(map.getZoom(), 12), { animate: true });
  };

  return (
    <div className="absolute right-3 bottom-[calc(96px+1rem)] flex flex-col gap-2 z-20">
      <Icon label="Zoom in" onClick={zoomIn}><Plus className="w-6 h-6" /></Icon>
      <Icon label="Zoom out" onClick={zoomOut}><Minus className="w-6 h-6" /></Icon>
      <Icon label="Recenter" onClick={recenter}><Crosshair className="w-6 h-6" /></Icon>
    </div>
  );
}

function Icon({ children, onClick, label }:{
  children: React.ReactNode; onClick?: () => void; label: string;
}) {
  return (
    <button
      aria-label={label}
      onClick={onClick}
      className="size-12 grid place-items-center rounded-2xl bg-white/90 text-slate-900 shadow-lg active:scale-95"
    >
      {children}
    </button>
  );
}

5) Patch MapView.tsx (Leaflet)

Make the map container relative, overlay the FAB, and wire the sheet.

// client/src/components/MapView.tsx (excerpt – key changes)
import React from "react";
import L from "leaflet";
import MapControlsFAB from "@/components/MapControlsFAB";
import { IncidentSheet, Incident } from "@/components/IncidentSheet";
import { Severity } from "@/lib/severity";
import "leaflet/dist/leaflet.css";

export default function MapView(/* your props */) {
  const mapEl = React.useRef<HTMLDivElement | null>(null);
  const mapRef = React.useRef<L.Map | null>(null);

  // incidents from your existing React Query / props
  const [incidents, setIncidents] = React.useState<Incident[]>([]);
  const [openSheet, setOpenSheet] = React.useState(false);
  const [activeSeverity, setActiveSeverity] = React.useState<Severity | "all">("all");
  const userCenterRef = React.useRef<[number, number] | null>(null);

  React.useEffect(() => {
    if (!mapEl.current || mapRef.current) return;

    const map = L.map(mapEl.current, { zoomControl: false }).setView([-22.56, 17.08], 13);
    mapRef.current = map;

    // keep your existing CartoDB tiles:
    L.tileLayer("https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png", {
      attribution: "&copy; OpenStreetMap & CARTO",
      maxZoom: 20,
    }).addTo(map);

    // TODO: keep your existing markers/clustering layer here

    // basic geolocate once (optional)
    if (navigator.geolocation) {
      navigator.geolocation.getCurrentPosition(
        (pos) => (userCenterRef.current = [pos.coords.longitude, pos.coords.latitude]),
        () => {}
      );
    }
  }, []);

  // When a list item is selected: center the map + (optionally) open a popup
  const handleSelectIncident = (id: string) => {
    const i = incidents.find(x => x.id === id);
    if (!i || !mapRef.current) return;
    mapRef.current.setView([i.lat, i.lng], 16, { animate: true });
  };

  return (
    <div className="relative w-full h-[70vh] rounded-2xl overflow-hidden"> {/* relative is important */}
      <div ref={mapEl} className="absolute inset-0" />

      {/* Floating buttons overlay */}
      <MapControlsFAB
        map={mapRef.current}
        getUserCenter={() => userCenterRef.current}
      />

      {/* Toggle button for bottom sheet (style like your circular FAB) */}
      <button
        onClick={() => setOpenSheet(true)}
        className="absolute left-3 bottom-[calc(96px+1rem)] z-20 h-12 px-4 rounded-2xl bg-white/90 text-slate-900 shadow-lg font-medium"
      >
        Incidents
      </button>

      {/* Bottom sheet with incidents */}
      <IncidentSheet
        open={openSheet}
        onOpenChange={setOpenSheet}
        incidents={incidents}
        activeSeverity={activeSeverity}
        setActiveSeverity={setActiveSeverity}
        onSelect={handleSelectIncident}
      />
    </div>
  );
}
