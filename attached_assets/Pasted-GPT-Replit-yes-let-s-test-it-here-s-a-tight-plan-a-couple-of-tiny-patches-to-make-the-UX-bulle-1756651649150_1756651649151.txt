4) Playwright E2E (copy-paste)

tests/report.desktop.spec.ts

import { test, expect } from "@playwright/test";

test.describe("Desktop tap-to-report", () => {
  test.use({ viewport: { width: 1366, height: 900 } });

  test("report flow opens sheet after map click", async ({ page }) => {
    await page.goto("/map");
    await page.getByRole("button", { name: "Report" }).click();

    // overlay visible
    await expect(page.getByText(/Click the map to set the incident location/i)).toBeVisible();

    // click roughly center of map
    const map = page.locator("#map");
    const box = await map.boundingBox();
    if (!box) throw new Error("Map not found");
    await page.mouse.click(box.x + box.width / 2, box.y + box.height / 2);

    // sheet opens and shows coords line
    await expect(page.getByText(/Location set:/i)).toBeVisible();
  });

  test("esc cancels report mode", async ({ page }) => {
    await page.goto("/map");
    await page.getByRole("button", { name: "Report" }).click();
    await expect(page.getByText(/Click the map/i)).toBeVisible();
    await page.keyboard.press("Escape");
    await expect(page.getByText(/Click the map/i)).toBeHidden();
  });

  test("mobile regression", async ({ page }) => {
    await page.setViewportSize({ width: 390, height: 844 });
    await page.goto("/map");
    await page.getByRole("button", { name: "Report" }).click();
    await expect(page.getByRole("button", { name: /Use my current location/i })).toBeVisible();
    await expect(page.getByText(/Click the map/i)).toHaveCount(0);
  });
});

5) Store unit tests (Vitest)

client/src/state/useQuickReportStore.test.ts

import { describe, it, expect, beforeEach } from "vitest";
import { useQuickReportStore } from "./useQuickReportStore";

const reset = () =>
  useQuickReportStore.setState(
    { sheetOpen: false, note: "", reportMode: false, categoryId: undefined, location: undefined, tempPin: undefined },
    true
  );

describe("useQuickReportStore desktop flow", () => {
  beforeEach(reset);

  it("start → confirm pin opens sheet with location", () => {
    const s = useQuickReportStore.getState();
    s.startReportMode();
    expect(useQuickReportStore.getState().reportMode).toBe(true);
    s.confirmTempPin({ lat: 1, lng: 2 });
    const st = useQuickReportStore.getState();
    expect(st.sheetOpen).toBe(true);
    expect(st.reportMode).toBe(false);
    expect(st.location).toEqual({ lat: 1, lng: 2 });
  });

  it("cancel clears temp pin", () => {
    const s = useQuickReportStore.getState();
    s.startReportMode();
    s.cancelReportMode();
    expect(useQuickReportStore.getState().tempPin).toBeUndefined();
  });
});
