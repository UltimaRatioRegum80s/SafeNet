// src/features/incidents/IncidentReportLayout.tsx
import { ReactNode } from "react";
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
import { Separator } from "@/components/ui/separator";
import { Button } from "@/components/ui/button";

type Props = {
  Form: ReactNode;        // your <IncidentReportForm />
  MapPanel: ReactNode;    // your <LocationPicker />
  Preview: ReactNode;     // optional live preview
  onSubmit?: () => void;  // wire to RHF handleSubmit if needed
  submitting?: boolean;
};

export default function IncidentReportLayout({ Form, MapPanel, Preview, onSubmit, submitting }: Props) {
  return (
    <div className="mx-auto w-full max-w-screen-xl px-3 py-4 md:py-6">
      {/* Mobile: single column with sticky submit */}
      <div className="md:hidden space-y-4">
        <Card className="shadow-sm">
          <CardHeader className="pb-2">
            <CardTitle className="text-lg">Report an Incident</CardTitle>
          </CardHeader>
          <CardContent className="space-y-4">{Form}</CardContent>
        </Card>

        {/* Map opens in a Sheet/Drawer on mobile – render a lightweight teaser or button here */}
        <Card className="shadow-sm">
          <CardHeader className="pb-2">
            <CardTitle className="text-base">Location</CardTitle>
          </CardHeader>
          <CardContent className="space-y-3">
            {MapPanel /* ideally a button to open <Sheet> with map to avoid heavy DOM on mobile */}
          </CardContent>
        </Card>

        {Preview ? (
          <Card className="shadow-sm">
            <CardHeader className="pb-2">
              <CardTitle className="text-base">Preview</CardTitle>
            </CardHeader>
            <CardContent>{Preview}</CardContent>
          </Card>
        ) : null}

        <div className="sticky bottom-2 z-40">
          <div className="rounded-2xl bg-background/70 backdrop-blur p-2 shadow-lg border">
            <Button className="w-full" onClick={onSubmit} disabled={submitting}>
              {submitting ? "Submitting…" : "Submit Incident"}
            </Button>
          </div>
        </div>
      </div>

      {/* Tablet / Desktop */}
      <div className="hidden md:grid md:grid-cols-2 lg:grid-cols-[1.2fr_0.8fr] gap-4">
        <Card className="shadow-sm">
          <CardHeader className="pb-2">
            <CardTitle className="text-xl">Report an Incident</CardTitle>
          </CardHeader>
          <CardContent className="space-y-4">{Form}</CardContent>
        </Card>

        <div className="space-y-4">
          <Card className="shadow-sm">
            <CardHeader className="pb-2">
              <CardTitle className="text-lg">Location</CardTitle>
            </CardHeader>
            <CardContent className="p-0">
              {/* ensure map container uses height constraints */}
              <div className="h-[320px] md:h-[380px] lg:h-[460px] rounded-b-2xl overflow-hidden">
                {MapPanel}
              </div>
            </CardContent>
          </Card>

          {Preview ? (
            <Card className="shadow-sm">
              <CardHeader className="pb-2">
                <CardTitle className="text-lg">Preview</CardTitle>
              </CardHeader>
              <CardContent>{Preview}</CardContent>
            </Card>
          ) : null}

          <Separator className="hidden lg:block" />
          <div className="hidden md:block">
            <Button className="w-full md:w-auto" onClick={onSubmit} disabled={submitting}>
              {submitting ? "Submitting…" : "Submit Incident"}
            </Button>
          </div>
        </div>
      </div>
    </div>
  );
}
