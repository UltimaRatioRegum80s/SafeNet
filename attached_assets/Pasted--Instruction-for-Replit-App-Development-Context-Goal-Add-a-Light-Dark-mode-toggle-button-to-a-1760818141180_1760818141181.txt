✅ Instruction for Replit / App Development Context

Goal:
Add a Light / Dark mode toggle button to all mobile pages (Home, Map, Report, Feed) that is sleek, semi-discreet, and visually modern, with a small pulse animation when toggled.

The toggle should float separately from the existing top floating navigation bar — not integrated into it — and the nav bar should be made slightly broader for aesthetic balance.

Safe-area aware:

right-4

Theme switch applies globally via Tailwind’s dark: classes.

Smooth animated transition and a gentle pulse ripple (Framer Motion) for user feedback.

Step 2: Create Theme Store
// src/store/useTheme.ts
import { create } from "zustand";

type Theme = "light" | "dark";
interface ThemeState {
  theme: Theme;
  toggleTheme: () => void;
}

export const useTheme = create<ThemeState>((set) => ({
  theme: (localStorage.getItem("theme") as Theme) || "dark",
  toggleTheme: () =>
    set((state) => {
      const next = state.theme === "dark" ? "light" : "dark";
      document.documentElement.classList.toggle("dark", next === "dark");
      localStorage.setItem("theme", next);
      return { theme: next };
    }),
}));

Step 3: Add Floating Theme Toggle with Pulse Animation
// src/components/ThemeToggle.tsx
import { Sun, Moon } from "lucide-react";
import { motion, AnimatePresence } from "framer-motion";
import { useTheme } from "@/store/useTheme";
import { cn } from "@/lib/utils";
import { useState } from "react";

export default function ThemeToggle() {
  const { theme, toggleTheme } = useTheme();
  const [pulse, setPulse] = useState(false);

  const handleToggle = () => {
    toggleTheme();
    setPulse(true);
    setTimeout(() => setPulse(false), 400); // reset pulse
  };

  return (
    <motion.button
      onClick={handleToggle}
      aria-label="Toggle theme"
      whileTap={{ scale: 0.9 }}
      className={cn(
        "fixed right-4 top-[calc(env(safe-area-inset-top)+10px)] z-60",
        "h-10 w-10 flex items-center justify-center rounded-full",
        "bg-neutral-900/70 dark:bg-neutral-100/40 backdrop-blur-md",
        "border border-white/10 shadow-[0_4px_12px_rgba(0,0,0,0.25)]",
        "transition-all duration-300 hover:scale-105 active:scale-95"
      )}
    >
      {/* Pulse effect */}
      {pulse && (
        <motion.span
          initial={{ scale: 0, opacity: 0.6 }}
          animate={{ scale: 2, opacity: 0 }}
          transition={{ duration: 0.6, ease: "easeOut" }}
          className="absolute inset-0 rounded-full bg-blue-400/30 dark:bg-yellow-300/30"
        />
      )}

      <AnimatePresence mode="wait">
        <motion.div
          key={theme}
          initial={{ rotate: 90, opacity: 0 }}
          animate={{ rotate: 0, opacity: 1 }}
          exit={{ rotate: -90, opacity: 0 }}
          transition={{ duration: 0.3 }}
        >
          {theme === "dark" ? (
            <Sun size={18} className="text-yellow-400" />
          ) : (
            <Moon size={18} className="text-blue-500" />
          )}
        </motion.div>
      </AnimatePresence>
    </motion.button>
  );
}


🪄 Effect:
When tapped, the icon rotates, the background pulse gently expands, and the app theme switches instantly with smooth fade transitions.
