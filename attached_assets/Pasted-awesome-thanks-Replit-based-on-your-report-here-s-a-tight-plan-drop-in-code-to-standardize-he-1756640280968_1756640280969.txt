1) TopBar.tsx

client/src/components/layout/TopBar.tsx

import * as React from "react";
import { cn } from "@/lib/utils";
import { Button } from "@/components/ui/button";
import {
  Share2, Search, Bell, MoreHorizontal, Home,
} from "lucide-react";

type Action = { key: string; aria: string; onClick?: () => void; icon?: React.ReactNode };

export interface TopBarProps {
  title: string;
  subtitle?: string;
  actions?: Action[];               // optional; defaults provided
  onMenu?: () => void;
  secondary?: React.ReactNode;      // tabs/chips row
  className?: string;
}

export function TopBar({
  title,
  subtitle,
  actions,
  onMenu,
  secondary,
  className,
}: TopBarProps) {
  const defaultActions: Action[] = [
    { key: "share", aria: "Share", icon: <Share2 className="h-5 w-5" /> },
    { key: "search", aria: "Search", icon: <Search className="h-5 w-5" /> },
    { key: "bell", aria: "Notifications", icon: <Bell className="h-5 w-5" /> },
    { key: "more", aria: "More", onClick: onMenu, icon: <MoreHorizontal className="h-5 w-5" /> },
  ];

  const items = actions ?? defaultActions;

  return (
    <header
      role="banner"
      className={cn(
        "sticky top-0 z-40 border-b bg-background/70 backdrop-blur supports-[backdrop-filter]:bg-background/60",
        className
      )}
    >
      <div className="mx-auto w-full max-w-screen-md px-3 md:px-4 h-14 md:h-16 flex items-center justify-between">
        <div className="flex items-center gap-2 min-w-0">
          <a href="/" aria-label="Home" className="shrink-0">
            <Home className="h-5 w-5 opacity-80" />
          </a>
          <div className="min-w-0">
            <div className="text-base font-semibold tracking-[-0.01em] truncate">
              {title}
            </div>
            {subtitle ? (
              <div className="hidden md:block text-xs text-muted-foreground truncate">
                {subtitle}
              </div>
            ) : null}
          </div>
        </div>

        <div className="flex items-center gap-1.5">
          {items.map((a) => (
            <Button
              key={a.key}
              variant="ghost"
              size="icon"
              className="h-9 w-9"
              aria-label={a.aria}
              onClick={a.onClick}
            >
              {a.icon}
            </Button>
          ))}
        </div>
      </div>

      {secondary ? (
        <div className="h-11 border-t bg-background/70 backdrop-blur px-2 overflow-x-auto no-scrollbar">
          <div className="mx-auto max-w-screen-md h-full flex items-center gap-2">
            {secondary}
          </div>
        </div>
      ) : null}
    </header>
  );
}

3) AppLayout.tsx

client/src/components/layout/AppLayout.tsx

import * as React from "react";
import { TopBar, TopBarProps } from "./TopBar";

interface AppLayoutProps {
  header: Pick<TopBarProps, "title" | "subtitle" | "actions" | "secondary">;
  children: React.ReactNode;
  footer?: React.ReactNode; // e.g., MobileFloatingTopMenu
  className?: string;
}

export default function AppLayout({ header, children, footer }: AppLayoutProps) {
  return (
    <div className="min-h-dvh bg-background text-foreground">
      <TopBar {...header} />
      <main className="mx-auto w-full max-w-screen-md px-3 md:px-4 py-3">
        {children}
      </main>
      {footer ? <div className="sticky bottom-0 z-30">{footer}</div> : null}
    </div>
  );
}

4) Wire it into Wouter

Update your root routing where you define <Route> components.

client/src/App.tsx (excerpt)

import { Route, Switch } from "wouter";
import AppLayout from "@/components/layout/AppLayout";
import { MobileFloatingTopMenu } from "@/components/MobileFloatingTopMenu"; // existing
import Dashboard from "@/pages/Dashboard";
import Chat from "@/pages/Chat";
import ReportIncident from "@/pages/report/ReportIncident";
import Map from "@/pages/community/Map";

export default function App() {
  return (
    <Switch>
      {/* Map keeps its own overlay header */}
      <Route path="/map" component={Map} />

      <Route path="/chat">
        <AppLayout
          header={{ title: "Chat", subtitle: "Neighborhood updates" }}
          footer={<MobileFloatingTopMenu />}
        >
          <Chat />
        </AppLayout>
      </Route>

      <Route path="/report">
        <AppLayout
          header={{
            title: "Report Incident",
            secondary: <IconTabs />, // optional; see below
          }}
          footer={<MobileFloatingTopMenu />}
        >
          <ReportIncident />
        </AppLayout>
      </Route>

      <Route path="/">
        <AppLayout
          header={{ title: "Dashboard", subtitle: "Community Management" }}
          footer={<MobileFloatingTopMenu />}
        >
          <Dashboard />
        </AppLayout>
      </Route>
    </Switch>
  );
}
