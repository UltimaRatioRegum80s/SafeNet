2) “Only show what’s on the map” (BBox view filter)
A) Hook to track bounds
// client/src/hooks/useLeafletBounds.ts
import { useEffect, useState } from "react";
import type { Map as LeafletMap, LatLngBounds } from "leaflet";

export type BBox = { north: number; south: number; east: number; west: number };

export default function useLeafletBounds(map: LeafletMap | null) {
  const [bbox, setBbox] = useState<BBox | null>(null);

  useEffect(() => {
    if (!map) return;
    const update = () => {
      const b: LatLngBounds = map.getBounds();
      setBbox({ north: b.getNorth(), south: b.getSouth(), east: b.getEast(), west: b.getWest() });
    };
    update();
    map.on("moveend zoomend", update);
    return () => { map.off("moveend zoomend", update); };
  }, [map]);

  return bbox;
}

B) Use it in the map page and filter client-side (no API change)
// client/src/pages/community/Map.tsx (excerpt)
import useLeafletBounds from "@/hooks/useLeafletBounds";

const bbox = useLeafletBounds(mapRef.current);

// derive visibleIncidents without touching your query/socket layers
const visibleIncidents = useMemo(() => {
  if (!bbox) return incidents;
  const { north, south, east, west } = bbox;
  return incidents.filter(i => {
    const { lat, lng } = i.position;
    return lat <= north && lat >= south && lng <= east && lng >= west;
  });
}, [incidents, bbox]);

// feed visibleIncidents to the sheet instead of full list
<IncidentSheet
  open={openSheet}
  onOpenChange={setOpenSheet}
  incidents={visibleIncidents}
  selectedId={selectedId}
  onSelect={/* same as above */}
/>
