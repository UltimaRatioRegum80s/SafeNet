// client/src/features/map/RecentIncidentsPanel.tsx
// (or wherever the panel lives)

export default function RecentIncidentsPanel() {
  const { incidents } = useFeedIncidents();

  return (
    <section className="card h-full overflow-hidden"> {/* contain everything */}
      {/* Sticky header */}
      <header className="sticky top-0 z-10 card-header/ghost bg-background/80 backdrop-blur supports-[backdrop-filter]:bg-background/60">
        <h3 className="text-lg font-semibold flex items-center gap-2">
          <span className="i-lucide-eye w-4 h-4" />
          Recent Incidents
        </h3>
      </header>

      {/* Scrollable content */}
      <div className="card-content p-3 pt-2 h-full overflow-y-auto pr-2 custom-scroll">
        {!incidents?.length ? (
          <div className="text-muted-foreground text-sm">No incidents.</div>
        ) : (
          <ul className="space-y-2">
            {incidents.slice(0, 50).map((i) => (
              <li key={i.id}>
                <article
                  className="rounded-2xl border border-white/5 bg-white/[.02]
                             hover:bg-white/[.04] hover:shadow-md
                             transition-colors duration-200
                             p-3"
                >
                  <div className="flex items-center gap-3">
                    {/* status dot */}
                    <span
                      className="size-2.5 rounded-full"
                      style={{ backgroundColor: severityDot(i.severity) }}
                      aria-hidden
                    />
                    {/* severity pill */}
                    <span
                      className={
                        "text-[10px] font-semibold px-2 py-0.5 rounded-full " +
                        severityPill(i.severity)
                      }
                    >
                      {i.severity}
                    </span>

                    <span className="ml-auto text-xs text-muted-foreground">
                      {timeAgo(i.createdAt)}
                    </span>
                  </div>

                  {i.title ? (
                    <h4 className="mt-2 text-sm font-medium leading-snug">
                      {i.title}
                    </h4>
                  ) : null}

                  {i.subtitle ? (
                    <p className="mt-1 text-xs text-muted-foreground line-clamp-2">
                      {i.subtitle}
                    </p>
                  ) : null}
                </article>
              </li>
            ))}
          </ul>
        )}
      </div>
    </section>
  );
}

/* tiny helpers — keep local to file or move to a utils file */
function timeAgo(iso: string) {
  const d = new Date(iso);
  const diff = (Date.now() - d.getTime()) / 1000;
  if (diff < 60) return `${Math.floor(diff)}s ago`;
  if (diff < 3600) return `${Math.floor(diff / 60)}m ago`;
  if (diff < 86400) return `${Math.floor(diff / 3600)}h ago`;
  return d.toLocaleDateString();
}
function severityDot(s: string) {
  switch (s) {
    case "CRITICAL": return "#ef4444"; // red-500
    case "HIGH":     return "#f59e0b"; // amber-500
    case "MEDIUM":   return "#facc15"; // yellow-400
    default:         return "#22c55e"; // green-500
  }
}
function severityPill(s: string) {
  switch (s) {
    case "CRITICAL": return "bg-red-500/15 text-red-400 ring-1 ring-red-500/20";
    case "HIGH":     return "bg-orange-500/15 text-orange-400 ring-1 ring-orange-500/20";
    case "MEDIUM":   return "bg-yellow-500/15 text-yellow-400 ring-1 ring-yellow-500/20";
    default:         return "bg-emerald-500/15 text-emerald-400 ring-1 ring-emerald-500/20";
  }
}
