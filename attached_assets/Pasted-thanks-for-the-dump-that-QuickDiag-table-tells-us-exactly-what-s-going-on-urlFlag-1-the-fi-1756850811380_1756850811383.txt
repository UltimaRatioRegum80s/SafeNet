// imports at top if not present
import { Flags } from '@/lib/flags';

// ... inside the effect where the Leaflet map instance is created:
if (Flags.fixMobileMarkers && mapRef.current) {
  // expose for console-only diagnostics
  (window as any)._map = mapRef.current;
}

// ... where you create/assign the MarkerClusterGroup:
if (!clusterRef.current) {
  clusterRef.current = L.markerClusterGroup(/* your existing options */);
}

// ensure the cluster group is actually on the map (no duplicates)
if (Flags.fixMobileMarkers && mapRef.current && clusterRef.current) {
  if (!mapRef.current.hasLayer(clusterRef.current)) {
    mapRef.current.addLayer(clusterRef.current);
  }
  // expose for diagnostics
  (window as any)._cluster = clusterRef.current;
}


Do not change any other logic. This doesn’t affect runtime behavior—it only exposes references when the flag is ON and makes sure the cluster is attached (idempotently).
