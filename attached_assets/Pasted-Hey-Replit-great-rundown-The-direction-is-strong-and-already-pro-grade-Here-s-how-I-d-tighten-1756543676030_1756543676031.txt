// client/src/state/incidentStore.ts
import { create } from "zustand";
import type { Severity } from "@/lib/severity";

type BBox = { north:number; south:number; east:number; west:number } | null;

type S = {
  selectedId: string|null;
  setSelectedId: (id:string|null)=>void;

  severity: Severity|"all";
  setSeverity: (s: S["severity"])=>void;

  timeRange: "all"|"24h"|"7d"|"30d";
  setTimeRange: (t:S["timeRange"])=>void;

  bbox: BBox;
  setBBox: (b:BBox)=>void;
};

export const useIncidentStore = create<S>((set)=>({
  selectedId: null, setSelectedId: (id)=>set({selectedId:id}),
  severity: "all", setSeverity: (s)=>set({severity:s}),
  timeRange: "all", setTimeRange: (t)=>set({timeRange:t}),
  bbox: null, setBBox: (b)=>set({bbox:b}),
}));

const { setBBox } = useIncidentStore();
const updateBBox = () => {
  const b = mapRef.current?.getBounds(); if (!b) return;
  setBBox({ north:b.getNorth(), south:b.getSouth(), east:b.getEast(), west:b.getWest() });
};
mapRef.current?.on("moveend zoomend", updateBBox);
updateBBox();
