2) De-dupe markers by id (prevents double counts from optimistic + socket)

In MapView.tsx (or your marker helper):

const seenIds = useRef<Set<string>>(new Set());

function addIncidentToCluster(inc: { id: string; type: string; latitude?: number; longitude?: number; lat?: number; lng?: number; title?: string; }) {
  if (!clusterRef.current) return;
  if (inc.id && seenIds.current.has(inc.id)) return; // de-dup
  if (inc.id) seenIds.current.add(inc.id);

  const lat = Number(inc.lat ?? inc.latitude);
  const lng = Number(inc.lng ?? inc.longitude);
  if (!Number.isFinite(lat) || !Number.isFinite(lng)) return;

  const ll = Math.abs(lat) > 90 && Math.abs(lng) <= 90 ? L.latLng(lng, lat) : L.latLng(lat, lng);
  const icon = getIncidentIcon?.(inc.type);
  const layer = icon
    ? L.marker(ll, { icon, title: inc.title ?? inc.type })
    : (L.circleMarker(ll, { radius: 7, color: "#2563eb", weight: 2, fillOpacity: 0.9 }) as unknown as L.Marker);

  (layer as any).incidentId = inc.id;
  clusterRef.current.addLayer(layer);

  if (showDebug) {
    const el = document.getElementById("nn-debug-count");
    if (el) el.textContent = `markers: ${clusterRef.current.getLayers().length}`;
  }
}

3) “In View” fetch that follows the camera (and auto-fallback)
Backend: add bbox filtering

server/index.ts

app.get("/api/incidents", async (req, res) => {
  const limit = Math.min(Number(req.query.limit) || 300, 1000);
  const bbox = (req.query.bbox as string | undefined)?.split(",").map(Number);
  const sinceHours = Number(req.query.sinceHours) || null;

  const whereClauses = [];
  if (bbox && bbox.length === 4 && bbox.every(Number.isFinite)) {
    const [minLng, minLat, maxLng, maxLat] = bbox;
    whereClauses.push(
      gte(incidents.longitude, minLng),
      lte(incidents.longitude, maxLng),
      gte(incidents.latitude,  minLat),
      lte(incidents.latitude,  maxLat),
    );
  }
  if (sinceHours) {
    const since = new Date(Date.now() - sinceHours * 3600_000);
    whereClauses.push(gte(incidents.createdAt, since));
  }

  const rows = await db
    .select()
    .from(incidents)
    .where(whereClauses.length ? and(...whereClauses) : undefined)
    .orderBy(desc(incidents.createdAt))
    .limit(limit);

  res.json(rows);
});

In MapView.tsx:

function bboxString(map: L.Map) {
  const b = map.getBounds();
  return [b.getWest(), b.getSouth(), b.getEast(), b.getNorth()].join(",");
}

// initial load
const fetchInView = async () => {
  const url = `/api/incidents?bbox=${bboxString(map)}&sinceHours=72`;
  const items = await fetch(url).then(r => r.json());
  items.forEach(addIncidentToCluster);
};
map.on("moveend zoomend", () => {
  if (currentScope === "inView") fetchInView();
});
