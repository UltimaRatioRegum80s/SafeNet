1) Deep-link open by type (wouter)

Open the panel when a type query param is present. Also update URL when user selects a type on the grid so you can share a link.

// client/src/pages/community/ReportIncident.tsx
import { useLocation } from "wouter";
import { useEffect } from "react";
import { useReportPanel } from "@/features/report/useReportPanel";
import { INCIDENT_TYPES } from "@/features/report/incidentTypes"; // if you’ve centralized it
import ReportPanel from "@/features/report/ReportPanel";

function getQuery(search: string) {
  const q = new URLSearchParams(search.replace(/^\?/, ""));
  return Object.fromEntries(q.entries());
}

export default function ReportIncidentPage() {
  const { openWith } = useReportPanel();
  const [path, setLocation] = useLocation();
  const search = typeof window !== "undefined" ? window.location.search : "";
  const query = getQuery(search);

  // open on load if ?type=<id>
  useEffect(() => {
    if (query.type) {
      const t = INCIDENT_TYPES.find(x => x.id === query.type);
      if (t) openWith(t);
    }
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [query.type]);

  function selectType(t: (typeof INCIDENT_TYPES)[number]) {
    const url = new URL(window.location.href);
    url.searchParams.set("type", t.id);
    // push new URL (no full reload)
    window.history.replaceState({}, "", url.toString());
    openWith(t);
  }

  return (
    <div className="relative mx-auto max-w-6xl p-4">
      <h1 className="text-2xl font-semibold mb-4">Quick Report</h1>

      <div className="grid grid-cols-2 sm:grid-cols-3 xl:grid-cols-4 gap-4">
        {INCIDENT_TYPES.map((t) => (
          <button
            key={t.id}
            className="rounded-2xl p-4 bg-slate-800/40 hover:bg-slate-800 transition shadow-sm border border-slate-700/40"
            onClick={() => selectType(t)}
          >
            <div className="flex flex-col items-center gap-3">
              <img src={t.iconSrc} alt={t.label} className="h-12 w-12" />
              <div className="text-sm text-center">{t.label}</div>
            </div>
          </button>
        ))}
      </div>

      <ReportPanel />
    </div>
  );
}

4) ReportPanel small a11y/UX polish

ESC/Back closes panel (shadcn handles ESC; we’ll also clear ?type).

Reduce motion respect.

Prevent scroll bleed when panel open.

// client/src/features/report/ReportPanel.tsx
import { useEffect } from "react";
import { useLocation } from "wouter";

// clear ?type on close
function clearTypeParam() {
  const url = new URL(window.location.href);
  url.searchParams.delete("type");
  window.history.replaceState({}, "", url.toString());
}

export default function ReportPanel() {
  // …existing code…

  // prevent background rubber-band scrolling on iOS while open
  useEffect(() => {
    if (open) {
      const prev = document.body.style.overscrollBehavior;
      document.body.style.overscrollBehavior = "contain";
      return () => { document.body.style.overscrollBehavior = prev; };
    }
  }, [open]);

  // respect reduced motion for any subtle animations you added
  const prefersReduced = window.matchMedia?.("(prefers-reduced-motion: reduce)")?.matches;

  return (
    <Sheet
      open={open}
      onOpenChange={(o) => {
        if (!o) {
          clearTypeParam();
          track("report_cancel", { type: selected?.id });
          close();
        }
      }}
    >
      {/* existing SheetContent … maybe gate any animated pieces on !prefersReduced */}
    </Sheet>
  );
}
