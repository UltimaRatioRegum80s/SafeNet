// server/util/sanitize.ts
export function sanitizePlainText(input: unknown, max = 1000) {
  const s = String(input ?? "").trim().slice(0, max);
  // Escape HTML entities so text is safe to render
  return s
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#39;");
}

import { isNull, and, eq } from "drizzle-orm";

router.get("/api/incidents/:id/comments", async (req, res) => {
  const { id } = req.params;
  const limit = Math.min(Number(req.query.limit ?? 20), 100);
  const offset = Math.max(Number(req.query.offset ?? 0), 0);

  const items = await db.query.incidentComments.findMany({
    where: and(
      eq(incidentComments.incidentId, id),
      isNull(incidentComments.deletedAt)             // ✅ hide deleted
    ),
    orderBy: (t, { desc }) => [desc(t.createdAt)],
    limit,
    offset,
  });

  res.json({ items, nextOffset: items.length === limit ? offset + limit : null });
});

3) Preload reviews count (visible before open) — without joining sockets for all cards

We’ll split the hook into fetch and live phases:

Fetch always (so the count shows).

Join socket only when the card is open (to avoid many subscriptions).

A) Update hook
// client/src/features/incidents/api.ts
export function useReviewSummary(
  incidentId: string,
  fetchEnabled: boolean,    // fetch count
  liveEnabled: boolean      // join socket for live updates
) {
  const key = ["incidents", incidentId, "reviews"];
  const q = useQuery({
    queryKey: key,
    enabled: fetchEnabled,
    queryFn: async () => {
      const r = await fetch(`/api/incidents/${incidentId}/reviews`);
      return (await r.json()) as { count: number; reviewedByMe: boolean };
    },
    staleTime: 15_000,
  });

  // Live updates only when panel is open
  React.useEffect(() => {
    if (!liveEnabled) return;
    const room = `incident:${incidentId}`;
    socket.emit("join", room);
    const handler = (p: any) => { if (p.incidentId === incidentId) q.refetch(); };
    socket.on("incident:review:count", handler);
    return () => {
      socket.emit("leave", room);
      socket.off("incident:review:count", handler);
    };
  }, [incidentId, liveEnabled]);

  return q;
}

// client/src/components/IncidentCard.tsx
