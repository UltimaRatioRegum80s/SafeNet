1) Neighborhood rooms: server wiring
a) Ensure sockets join their neighborhood room on connect

Use session/auth to determine the user’s neighborhoodId and join nh:<id>.

// server/realtime.ts
import type { Server as HttpServer } from "http";
import { Server } from "socket.io";
import type { IncomingMessage } from "http";

export let io: Server;

export function initRealtime(server: HttpServer, getSessionForSocket: (req: IncomingMessage) => Promise<{ userId?: string; neighborhoodId?: string; isAdmin?: boolean }>) {
  io = new Server(server, {
    cors: { origin: true, credentials: true },
  });

  io.on("connection", async (socket) => {
    try {
      const session = await getSessionForSocket(socket.request);
      const nhId = session?.neighborhoodId;
      if (nhId) {
        socket.join(`nh:${nhId}`);
      }

      // Optional manual join (fallback if you want client-initiated):
      socket.on("join-neighborhood", (payload: { neighborhoodId: string }) => {
        if (payload?.neighborhoodId) socket.join(`nh:${payload.neighborhoodId}`);
      });
    } catch (e) {
      // no-op; user remains in public room
    }
  });

  return io;
}
