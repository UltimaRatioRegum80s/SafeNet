A) Robust timeAgo (seconds or ms)
// client/src/lib/timeAgo.ts
export function timeAgo(input?: number | string | Date) {
  if (input == null) return "Just now";
  let d: Date;
  if (input instanceof Date) d = input;
  else if (typeof input === "number") d = new Date(input < 1e12 ? input * 1000 : input);
  else d = new Date(input);
  if (Number.isNaN(d.getTime())) return "Unknown time";

  const s = Math.floor((Date.now() - d.getTime()) / 1000);
  if (s < 60) return `${s}s ago`;
  const m = Math.floor(s / 60);
  if (m < 60) return `${m}m ago`;
  const h = Math.floor(m / 60);
  if (h < 24) return `${h}h ago`;
  const dys = Math.floor(h / 24);
  return `${dys}d ago`;
}


If you prefer date-fns:

import { formatDistanceToNow } from "date-fns";
export const timeAgoDF = (input?: number|string|Date) => {
  if (input == null) return "Just now";
  const n = typeof input === "number" ? (input < 1e12 ? input * 1000 : input) : input;
  const d = new Date(n as any);
  if (Number.isNaN(d.getTime())) return "Unknown time";
  return formatDistanceToNow(d, { addSuffix: true });
};

B) Severity atoms for lists
// client/src/components/SeverityUI.tsx
import { AlertTriangle, MapPin, Siren, HandHelping } from "lucide-react";

export type Severity = "low" | "medium" | "high" | "emergency";

export const severityColorClass = (s: Severity) =>
  ({ low: "bg-emerald-500", medium: "bg-amber-500", high: "bg-orange-500", emergency: "bg-red-600" }[s]);

export function SeverityChip({ sev }: { sev: Severity }) {
  return (
    <span className="px-2 py-0.5 rounded-full bg-white/10 border border-white/10 text-[10px] font-semibold">
      {sev.toUpperCase()}
    </span>
  );
}

export function SeverityIcon({ sev, className = "w-4 h-4 text-white/80" }:{ sev: Severity; className?: string }) {
  switch (sev) {
    case "emergency": return <Siren className={className} />;
    case "high":      return <AlertTriangle className={className} />;
    case "medium":    return <MapPin className={className} />;
    default:          return <HandHelping className={className} />;
  }
}

C) Shared IncidentList (reused in Sidebar & Sheet)
// client/src/components/IncidentList.tsx
import React from "react";
import { timeAgo } from "@/lib/timeAgo";
import { SeverityIcon, SeverityChip, severityColorClass } from "@/components/SeverityUI";
import { useIncidentStore } from "@/state/useIncidentStore";

export interface IncidentMarker {
  id: string;
  type: "crime" | "safety_alert" | "suspicious" | "emergency";
  title: string;
  description: string;
  position: { lat: number; lng: number };
  severity: "low" | "medium" | "high" | "emergency";
  created_at: number; // secs or ms
}

export default function IncidentList({
  items,
  onSelect,
  dense = false,
}: {
  items: IncidentMarker[];
  onSelect: (id: string) => void;
  dense?: boolean;
}) {
  const selectedId = useIncidentStore((s) => s.selectedId);

  return (
    <div className={`px-2 pb-3 overflow-y-auto ${dense ? "space-y-1" : ""}`}>
      {items.map((i) => {
        const isActive = i.id === selectedId;
        return (
          <button
            key={i.id}
            onClick={() => onSelect(i.id)}
            aria-selected={isActive}
            className={`w-full text-left ${dense ? "p-2" : "p-3"} mb-2 rounded-2xl border transition
              ${isActive ? "bg-white/15 border-white/30" : "bg-white/5 border-white/10 hover:bg-white/10"}
            `}
          >
            <div className="flex items-start gap-3">
              <div className={`mt-1 size-3 rounded-full ${severityColorClass(i.severity)}`} />
              <div className="flex-1 min-w-0">
                <div className="flex items-center gap-2">
                  <SeverityIcon sev={i.severity} />
                  <div className="font-medium truncate">{i.title}</div>
                </div>
                {i.description && (
                  <div className="text-sm text-white/70 line-clamp-2">{i.description}</div>
                )}
                <div className="mt-1 flex items-center gap-2 text-xs text-white/60">
                  <SeverityChip sev={i.severity} />
                  <span>{timeAgo(i.created_at)}</span>
                </div>
              </div>
            </div>
          </button>
        );
      })}
      {items.length === 0 && (
        <div className="p-6 text-center text-white/60">No incidents.</div>
      )}
    </div>
  );
}

D) Map ↔ List sync handler (center + select)
// client/src/pages/community/Map.tsx (excerpt)
import { useIncidentStore } from "@/state/useIncidentStore";
import type { IncidentMarker } from "@/components/IncidentList";

const MIN_SELECT_ZOOM = 15;

function useSelectAndCenter(mapRef: React.MutableRefObject<L.Map | null>, incidents: IncidentMarker[]) {
  const setSelectedId = useIncidentStore((s) => s.setSelectedId);

  return (id: string) => {
    setSelectedId(id);
    const i = incidents.find((x) => x.id === id);
    const map = mapRef.current;
    if (!i || !map) return;
    map.setView([i.position.lat, i.position.lng], Math.max(MIN_SELECT_ZOOM, map.getZoom()), { animate: true });
  };
}

G) Fix bbox init error (guard until ready)
// after map init
function wireBBox(map: L.Map) {
  let t: any;
  const push = () => {
    try {
      if (!(map as any)._loaded) return;
      const b = map.getBounds();
      useIncidentStore.getState().setBBox({
        north: b.getNorth(), south: b.getSouth(), east: b.getEast(), west: b.getWest(),
      });
    } catch { /* transient during init */ }
  };
  const debounced = () => { clearTimeout(t); t = setTimeout(push, 180); };

  map.whenReady(push);
  map.on("load", debounced);
  map.on("moveend", debounced);
  map.on("zoomend", debounced);
}
