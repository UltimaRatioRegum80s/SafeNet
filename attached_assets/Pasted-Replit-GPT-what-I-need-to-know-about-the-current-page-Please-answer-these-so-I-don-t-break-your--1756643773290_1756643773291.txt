src/components/SeverityButtons.tsx

import clsx from "clsx";
import { useId } from "react";

export type Severity = "low" | "medium" | "high" | "critical";

type Props = {
  value: Severity | null;
  onChange: (v: Severity) => void;
  className?: string;
  immediateNav?: boolean; // if true, we’ll call onChange and expect page to navigate
};

const baseNeutral =
  "h-14 w-full rounded-full font-semibold text-base transition-all " +
  "bg-slate-200/85 text-slate-900 shadow-sm " +
  "focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-offset-2 focus-visible:ring-offset-slate-950";

export default function SeverityButtons({
  value,
  onChange,
  className,
  immediateNav = false,
}: Props) {
  const id = useId();
  const handle = (v: Severity) => {
    if (navigator.vibrate) navigator.vibrate(8);
    onChange(v);
  };

  return (
    <div
      role="radiogroup"
      aria-labelledby={`${id}-label`}
      className={clsx("w-full max-w-sm space-y-4", className)}
    >
      <button
        role="radio"
        aria-checked={value === "low"}
        className={clsx(baseNeutral, "ring-2 ring-yellow-400")}
        onClick={() => handle("low")}
      >
        Low
      </button>

      <button
        role="radio"
        aria-checked={value === "medium"}
        className={clsx(baseNeutral, "ring-2 ring-amber-600")}
        onClick={() => handle("medium")}
      >
        Medium
      </button>

      <button
        role="radio"
        aria-checked={value === "high"}
        className={clsx(baseNeutral, "ring-2 ring-red-500")}
        onClick={() => handle("high")}
      >
        High
      </button>

      <button
        role="radio"
        aria-checked={value === "critical"}
        className={clsx(
          "h-14 w-full rounded-full font-semibold text-base transition-all",
          "bg-red-700 text-white shadow-md",
          "focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-red-400",
          "focus-visible:ring-offset-2 focus-visible:ring-offset-slate-950"
        )}
        onClick={() => handle("critical")}
      >
        Critical
      </button>
    </div>
  );
}
